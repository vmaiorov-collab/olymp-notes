Лекция: структуры данных на дереве отрезков

Запись начинается с настройки звука и разговоров до начала; содержательная часть стартует в . Лекция посвящена «необычным» вариантам дерева отрезков и родственным структурам: массовые операции, дерево Фенвика, дерево отрезков-сортировок (merge sort tree) с каскадированием, а затем серия задач на размерности — от одномерной до четырёхмерной. Основной лектор — первый преподаватель (в программе курса — Данис Валиуллин); ведущие занятия по ходу задают друг другу вопросы и спорят с залом, эти обсуждения передаются сжато.

Дерево отрезков с массовыми операциями

Задача. На массиве нужны две операции: += x на отрезке и сумма на отрезке (–). Дерево отрезков с суммами строим как обычно: в каждой вершине хранится сумма на её отрезке, листья — элементы, корень — весь массив (–). Запрос суммы остаётся прежним; меняется только обновление (–).

Любой отрезок $[l,r]$ распадается на $O(\log n)$ полностью накрытых вершин. Если ко всем элементам вершины с $s$ листьями прибавить $x$, её сумма вырастет на $s\cdot x$ (–).

Почему нельзя просто обновить накрытые вершины и их предков. Тогда запрос по корню и по накрытым вершинам будет верным, но если спуститься ниже накрытой вершины, дети ничего не знают о прибавлении: информация о $x$ «потеряна» (–).

Ленивые операции. В каждой вершине храним два числа: сумму $\mathrm{sum}_v$ и отложенное прибавление $\mathrm{add}_v$ — сколько ещё нужно добавить сыновьям (–). Когда накрытая вершина получает $+x$, мы сразу правильно меняем её сумму ($+\,x\cdot\text{длина}$) и записываем $x$ в $\mathrm{add}$ обоих сыновей — а дальше «они уже сами разберутся» (–). Когда мы приходим в сына с ненулевым $\mathrm{add}$, ко всем листьям его поддерева нужно добавить $\mathrm{add}$, поэтому сумма растёт на $\mathrm{add}\cdot\text{длина}$, а сыновьям передаётся то же значение (–).

Это оформляется функцией push(v): прибавить $\mathrm{add}_v$ сыновьям, пересчитать сумму, обнулить $\mathrm{add}_v$ (–). Она ничего не портит: повторный вызов не изменит значений (–). Правило: вызывать push перед любым спуском вниз — и в запросе суммы, и в обновлении (–).

struct LazySum {
    int n;
    vector<ll> sum, add;                         // add[v] — сколько ещё нужно добавить сыновьям
    LazySum(int n) : n(n), sum(4 * n, 0), add(4 * n, 0) {}
    void apply(int v, int l, int r, ll x) { sum[v] += x * (r - l); add[v] += x; }
    void push(int v, int l, int r) {             // безвредно вызывать сколько угодно раз
        if (!add[v]) return;
        int m = (l + r) / 2;
        apply(2 * v, l, m, add[v]); apply(2 * v + 1, m, r, add[v]);
        add[v] = 0;
    }
    void update(int v, int l, int r, int ql, int qr, ll x) {   // [ql, qr) += x
        if (qr <= l || r <= ql) return;
        if (ql <= l && r <= qr) { apply(v, l, r, x); return; }
        push(v, l, r);
        int m = (l + r) / 2;
        update(2 * v, l, m, ql, qr, x); update(2 * v + 1, m, r, ql, qr, x);
        sum[v] = sum[2 * v] + sum[2 * v + 1];
    }
    ll get(int v, int l, int r, int ql, int qr) {
        if (qr <= l || r <= ql) return 0;
        if (ql <= l && r <= qr) return sum[v];
        push(v, l, r);
        int m = (l + r) / 2;
        return get(2 * v, l, m, ql, qr) + get(2 * v + 1, m, r, ql, qr);
    }
    void update(int l, int r, ll x) { update(1, 0, n, l, r, x); }
    ll get(int l, int r) { return get(1, 0, n, l, r); }
};

Площадь объединения прямоугольников

Задача. На плоскости $n$ прямоугольников со сторонами, параллельными осям (они могут пересекаться и быть вложены); найти площадь их объединения. Ограничения: координаты до $10^9$, $n\le 10^5$ (–). Зал отмечает, что задача известная («баян»).

Сжатие по $y$. Оставим только те значения $y$, которые встречаются среди границ прямоугольников. Промежутки между соседними значениями — «полоски», у каждой есть вес (её длина), и площадь считается как сумма весов покрытых полосок на ширину (–). Различных полосок не больше $2n$, поэтому по ним можно строить дерево отрезков с массовыми операциями (–).

Заметающая прямая. Выпишем события по $x$ в один массив: «прямоугольник открылся» ($+1$ на отрезке полосок) и «закрылся» ($-1$). При равных $x$ сначала открытия, затем закрытия (–). Значение в полоске — сколько прямоугольников её сейчас накрывают, оно никогда не уходит ниже нуля, потому что закрытие идёт после открытия (–).

Между двумя соседними значениями $x_10)\cdot(x_2-x_1)$ (–). Нужна структура с операциями «прибавить на отрезке» и «вес позиций, не равных нулю» (–). Так как все значения неотрицательны, ненулевые — это «все, кроме нулевых», а нулевые — это минимальные, если минимум равен нулю. Поэтому в вершине храним минимум и суммарный вес позиций, где он достигается (–). Пересчёт вершины: если у сыновей минимумы равны — веса складываем, иначе берём меньший минимум с его весом (–); прибавление на всей вершине сдвигает минимум и не меняет количества (–).

Покрытая длина $=\text{общая длина}-(\text{вес минимумов, если минимум}=0)$, добавляем её, умноженную на $x_2-x_1$, к площади (–).

struct MinCount {
    int n;
    vector<ll> mn, add;
    vector<ll> cnt;                              // суммарный вес позиций, где достигается минимум
    MinCount(const vector<ll>& w) : n(w.size()), mn(4 * w.size(), 0), add(4 * w.size(), 0), cnt(4 * w.size(), 0) { build(1, 0, n, w); }
    void build(int v, int l, int r, const vector<ll>& w) {
        if (r - l == 1) { cnt[v] = w[l]; return; }
        int m = (l + r) / 2;
        build(2 * v, l, m, w); build(2 * v + 1, m, r, w);
        pull(v);
    }
    void pull(int v) {
        mn[v] = min(mn[2 * v], mn[2 * v + 1]);
        cnt[v] = (mn[2 * v] == mn[v] ? cnt[2 * v] : 0) + (mn[2 * v + 1] == mn[v] ? cnt[2 * v + 1] : 0);
    }
    void apply(int v, ll x) { mn[v] += x; add[v] += x; }
    void push(int v) { if (add[v]) { apply(2 * v, add[v]); apply(2 * v + 1, add[v]); add[v] = 0; } }
    void update(int v, int l, int r, int ql, int qr, ll x) {
        if (qr <= l || r <= ql) return;
        if (ql <= l && r <= qr) { apply(v, x); return; }
        push(v);
        int m = (l + r) / 2;
        update(2 * v, l, m, ql, qr, x); update(2 * v + 1, m, r, ql, qr, x);
        pull(v);
    }
    pair<ll, ll> query(int v, int l, int r, int ql, int qr) {      // (минимум, вес минимумов)
        if (qr <= l || r <= ql) return {LLONG_MAX, 0};
        if (ql <= l && r <= qr) return {mn[v], cnt[v]};
        push(v);
        int m = (l + r) / 2;
        auto a = query(2 * v, l, m, ql, qr), b = query(2 * v + 1, m, r, ql, qr);
        if (a.first < b.first) return a;
        if (b.first < a.first) return b;
        return {a.first, a.second + b.second};
    }
    void update(int l, int r, ll x) { update(1, 0, n, l, r, x); }
    pair<ll, ll> query(int l, int r) { return query(1, 0, n, l, r); }
};
ll union_area(const vector<array<int, 4>>& rect) {            // {x1, y1, x2, y2}, x1 < x2, y1 < y2
    vector<int> ys;
    for (auto& r : rect) { ys.push_back(r[1]); ys.push_back(r[3]); }
    sort(ys.begin(), ys.end()); ys.erase(unique(ys.begin(), ys.end()), ys.end());
    int m = (int)ys.size() - 1;                                // полоса i = [ys[i], ys[i+1]) с весом длины
    if (m <= 0) return 0;
    vector<ll> w(m);
    for (int i = 0; i < m; i++) w[i] = ys[i + 1] - ys[i];
    ll total = ys.back() - ys[0];
    MinCount tree(w);
    vector<array<int, 4>> ev;                                  // {x, -тип, l, r}: открытие (+1) идёт раньше закрытия
    for (auto& r : rect) {
        int l = lower_bound(ys.begin(), ys.end(), r[1]) - ys.begin();
        int h = lower_bound(ys.begin(), ys.end(), r[3]) - ys.begin();
        ev.push_back({r[0], -1, l, h}); ev.push_back({r[2], 1, l, h});
    }
    sort(ev.begin(), ev.end());
    ll area = 0;
    for (size_t i = 0; i < ev.size(); i++) {
        if (i) {
            auto [mn, c] = tree.query(0, m);
            ll covered = total - (mn == 0 ? c : 0);            // ненулевые = всё минус нули
            area += covered * (ev[i][0] - ev[i - 1][0]);
        }
        tree.update(ev[i][2], ev[i][3], -ev[i][1]);            // -(-1) = +1 при открытии, -1 при закрытии
    }
    return area;
}

Обсуждение: «снизу», неявное дерево, Фенвик

По ходу зал спорит о нерекурсивном («снизу») дереве отрезков: оно пишется быстрее, чем «сверху», а при должном желании на нём можно сделать и массовые операции, но это сложно (–). В лекции задачи решаются «сверху».

Неявное (динамическое) дерево отрезков. Нужно дерево на отрезке размера до $10^{18}$, при том что запросов всего порядка $10^5$ (–). У вершины левый и правый сыновья могут быть не созданы (индекс $-1$); значение в несозданной вершине считается значением «по умолчанию» (нули или арифметическая прогрессия — что-нибудь, что легко посчитать без хранения) (–). При обновлении, если нужного сына нет, создаём его и идём дальше; при запросе — аналогично или просто считаем, что там нули (–). Запрос затрагивает $\log C\approx 60$ вершин, всего создаётся порядка $60Q$ вершин — гораздо меньше $10^{18}$ (–).

struct DynTree {
    struct Node { ll sum = 0; int l = -1, r = -1; };
    vector<Node> t; ll N;
    DynTree(ll N) : N(N) { t.emplace_back(); }
    int child(int v, bool right) {
        int c = right ? t[v].r : t[v].l;
        if (c < 0) { c = t.size(); t.emplace_back(); (right ? t[v].r : t[v].l) = c; }
        return c;
    }
    void add(ll pos, ll x) {                                   // a[pos] += x, создаём только вершины на пути
        int v = 0; ll lo = 0, hi = N;
        t[v].sum += x;
        while (hi - lo > 1) {
            ll m = (lo + hi) / 2;
            bool right = pos >= m;
            v = child(v, right);
            (right ? lo : hi) = m;
            t[v].sum += x;
        }
    }
    ll get(int v, ll lo, ll hi, ll l, ll r) {                  // сумма на [l, r)
        if (v < 0 || r <= lo || hi <= l) return 0;             // отсутствующая вершина — нули
        if (l <= lo && hi <= r) return t[v].sum;
        ll m = (lo + hi) / 2;
        return get(t[v].l, lo, m, l, r) + get(t[v].r, m, hi, l, r);
    }
    ll get(ll l, ll r) { return get(0, 0, N, l, r); }
};

Дерево Фенвика

Определение. В ячейке $i$ дерева Фенвика хранится сумма элементов исходного массива на промежутке от $i$ вниз, не включая $i-2^k$, где $2^k$ — младший единичный бит числа $i$: для $i=8$ это сумма до нуля, для $i=3$ — до двойки (–). Ограничение по сути одно: нужна обратимая операция (сумма подходит), потому что запрос на отрезке сводится к двум префиксам: $\mathrm{sum}(l,r)=\mathrm{pref}(r)-\mathrm{pref}(l-1)$ (–).

Уточнение. В записи лектор говорит «максимальный бит», но его же примеры ($8\to 0$, $3\to 2$) соответствуют младшему единичному биту, то есть $i\,\&\,(-i)$; именно так определено дерево Фенвика.
  • Префикс $i$: берём ячейку $i$ (она закрывает промежуток вниз до $i-2^k$), затем ячейку $i-2^k$ и так далее, каждый раз убирая один единичный бит — всего $O(\log n)$ слагаемых (–).
  • Прибавление в точке $i$: нужно обновить ячейки, чей промежуток содержит $i$; переходим к ним, каждый раз прибавляя младший бит: $i \mathrel{+}= i\,\&\,(-i)$ (–).

Код — две короткие функции: for (; i <= n; i += i & -i) t[i] += x и for (; i > 0; i -= i & -i) ans += t[i]. Формулу $i\,\&\,(-i)$ можно просто запомнить: она работает потому, что числа в компьютере хранятся в дополнительном коде, где $-1$ — все единицы (–).

Построение — заполнить нулями и сделать $n$ обновлений, то есть $O(n\log n)$. Лектор в споре настаивает, что это нормально (запросы асимптотически логарифмические, а на степенях двойки работают даже быстрее), хотя за $O(n)$ тоже можно (–). Дерево Фенвика быстрее и занимает вчетверо меньше памяти, чем «дерево отрезков на $4n$» (–).

struct Fenwick {
    int n; vector<ll> t;
    Fenwick(int n) : n(n), t(n + 1, 0) {}
    void add(int i, ll x) { for (; i <= n; i += i & -i) t[i] += x; }   // индексы с 1
    ll pref(int i) { ll r = 0; for (; i > 0; i -= i & -i) r += t[i]; return r; }
    int kth(ll k) {                                                     // минимальное i с pref(i) >= k
        int pos = 0;
        for (int pw = 1 << __lg(n); pw; pw >>= 1)
            if (pos + pw <= n && t[pos + pw] < k) { pos += pw; k -= t[pos]; }
        return pos + 1;
    }
};

Дерево отрезков-сортировок (merge sort tree)

Идея. В каждой вершине дерева хранить не сумму, а весь отсортированный набор элементов её отрезка: в корне — отсортированный массив, в сыновьях — отсортированные половины. Это ровно то, что строит сортировка слиянием: рекурсивные вызовы возвращают отсортированные массивы, которые мы сохраняем (–).

  • Память. Каждый элемент лежит на $\log n$ слоях, итого $O(n\log n)$ (–).
  • Построение. $O(n\log n)$, как у сортировки слиянием (–).
  • Запрос «сколько чисел меньше $x$ на $[l,r]$». Разбиваем отрезок на $O(\log n)$ вершин, в каждой делаем lower_bound и складываем: $O(\log^2 n)$ (–).
  • Обновление. Если вершины хранят vector, удалить и добавить элемент долго. Заменим векторы на ordered set из pb_ds — обычное множество, у которого есть операции «номер ключа» (order_of_key) и «ключ по номеру» (find_by_order); тогда обновление и запрос работают за $O(\log^2 n)$ (–). Лектор советует просто запомнить подключение (ext/pb_ds/assoc_container.hpp, tree_policy.hpp, пространство имён __gnu_pbds) и признаёт, что оно работает примерно в четыре раза дольше, чем собственное декартово дерево (–).

Частичное каскадирование

Теперь о том, что в зале, по признанию слушателей, не знает никто. Цель: убрать лишний логарифм из запроса «сколько чисел меньше $x$ на отрезке» в дереве отрезков-сортировок. Обновлений при этом нет — структура строится один раз (–). Лектор оговаривает, что на практике оно часто работает не быстрее обычного дерева сортировок: хотя асимптотика на $\log$ лучше, память в 4 раза больше, много кэш-промахов; но приводит замер на сабмитах: без каскадирования около 1,91 с, с каскадированием 1,37 с. Конечно, всё зависит от качества реализации (–).

Идея. Массив корня получен слиянием массивов двух сыновей (двигаем два указателя, берём меньшее). Сделаем в корне один бинпоиск и найдём позицию $p$ первого элемента, не меньшего $x$ (–). Утверждение: для этого же запроса позиция в каждом сыне определяется по $p$ без бинпоиска:

  • элемент, найденный в корне, пришёл из одной из половин; в этой половине он же будет первым не меньшим $x$ (если он первый на всём отрезке, то и в своей половине тоже) (–);
  • в другой половине первым не меньшим $x$ будет ближайший справа от $p$ элемент, пришедший оттуда (–).

Значит, для каждой позиции $i$ массива вершины достаточно заранее посчитать за линию, сколько из первых $i$ элементов пришло из левого сына (в левом сыне это его позиция, в правом — $i$ минус это число) — «следующий пришедший из другого массива» (–). Тогда один бинпоиск в корне и дальше обычный спуск по дереву, в котором в каждой вершине позиция известна «с неба»: $O(\log n)$ на запрос вместо $O(\log^2 n)$ (–). Память — $O(n\log n)$ с большим множителем (лектор говорит про «четыре» (–)).

На вопрос из зала («а если сверху есть ещё элементы между») лектор отвечает: лишние элементы можно просто игнорировать — бинпоиск в корне указывает куда-то в «сжатый» массив; нужен лишь «следующий неравный», он считается предподсчётом (–).

struct MergeSortTree {
    int n; vector<vector<int>> val, toL;          // toL[v][i]: сколько из первых i элементов val[v] лежат в левом сыне
    MergeSortTree(const vector<int>& a) : n(a.size()), val(4 * a.size()), toL(4 * a.size()) { build(1, 0, n, a); }
    void build(int v, int l, int r, const vector<int>& a) {
        if (r - l == 1) { val[v] = {a[l]}; return; }
        int m = (l + r) / 2;
        build(2 * v, l, m, a); build(2 * v + 1, m, r, a);
        auto &L = val[2 * v], &R = val[2 * v + 1];
        val[v].resize(r - l); toL[v].assign(r - l + 1, 0);
        size_t i = 0, j = 0;
        for (int k = 0; k < r - l; k++) {
            bool left = j == R.size() || (i < L.size() && L[i] <= R[j]);
            val[v][k] = left ? L[i++] : R[j++];
            toL[v][k + 1] = toL[v][k] + left;
        }
    }
    int cnt(int v, int l, int r, int ql, int qr, int p) {   // p — сколько элементов вершины v меньше x
        if (qr <= l || r <= ql || p == 0) return 0;
        if (ql <= l && r <= qr) return p;

        return cnt(2 * v, l, m, ql, qr, toL[v][p]) + cnt(2 * v + 1, m, r, ql, qr, p - toL[v][p]);
    }
    int count_less(int ql, int qr, int x) {                 // на [ql, qr): один бинпоиск в корне, дальше O(1) на вершину
        int p = lower_bound(val[1].begin(), val[1].end(), x) - val[1].begin();
        return cnt(1, 0, n, ql, qr, p);
    }
};

// k-я порядковая статистика на [l, r): бинпоиск по ответу, O(log^2)
int kth_on_range(MergeSortTree& t, int l, int r, int k) {   // k с 1
    auto& all = t.val[1];
    int lo = 0, hi = (int)all.size() - 1;
    while (lo < hi) {
        int mid = (lo + hi) / 2;
        if (t.count_less(l, r, all[mid] + 1) >= k) hi = mid; else lo = mid + 1;
    }
    return all[lo];
}

Задача. Даны $n$ троек $(a_i,b_i,c_i)$. Тройка $i$ бесполезна, если существует $j$ с $a_j>a_i$, $b_j>b_i$, $c_j>c_i$. Сколько троек остаётся (–)? Двумерный случай. Отсортируем по $a$ по убыванию и пройдём, поддерживая максимум $b$ среди уже просмотренных: тройка (пара) бесполезна, если этот максимум строго больше её $b$ (–). Лектор записывает: сортировка по первому параметру превращает его «во время».

Три измерения. После сортировки по $a$ у нас остаются пары $(b,c)$, идём по ним справа налево (в порядке, в котором нас могут убить) и складываем во вспомогательное дерево: индекс — $b$, значение — максимум $c$. Для очередной пары спрашиваем максимум $c$ на суффиксе по $b$ (тех, у кого $b$ больше): если он больше нашего $c$ — нас убивают (–). Нужно именно число убитых, а не число пар «убийца—жертва», поэтому достаточно максимума (–). Время $O(n\log n)$.

vector<char> dominated_3d(const vector<array<int, 3>>& p) {
    int n = p.size();
    vector<int> bs;
    for (auto& x : p) bs.push_back(x[1]);
    sort(bs.begin(), bs.end()); bs.erase(unique(bs.begin(), bs.end()), bs.end());
    int m = bs.size();
    FenwickMax f(m);                                // по «перевёрнутому» b хранит максимум c: суффикс по b
    auto rb = [&](int b) { return m - (int)(lower_bound(bs.begin(), bs.end(), b) - bs.begin()); };   // 1..m, больше b — меньше номер
    vector<int> ord(n); iota(ord.begin(), ord.end(), 0);
    sort(ord.begin(), ord.end(), [&](int x, int y) { return p[x][0] > p[y][0]; });
    vector<char> dead(n, 0);
    for (int i = 0; i < n;) {
        int j = i;
        while (j < n && p[ord[j]][0] == p[ord[i]][0]) j++;
        for (int k = i; k < j; k++) {               // сначала все запросы группы: равные a друг друга не убивают
            int id = ord[k];
            if (f.get(rb(p[id][1]) - 1) > p[id][2]) dead[id] = 1;   // есть b' > b с c' > c (c хранится +1, 0 = пусто)
        }
        for (int k = i; k < j; k++) { int id = ord[k]; f.upd(rb(p[id][1]), p[id][2]); }
        i = j;
    }
    return dead;
}

Четыре измерения. Попытки обойтись «по очереди» сжатием по каждой из переменных не работают (–): «обработать сколько-то параметров по отдельности и потом добавить ещё один» нельзя, нужно искать позицию, у которой все четыре больше, сразу (–). Многомерные Фенвики дают $\log^{d}$ и «можно сколько угодно измерений», но лектор утверждает, что можно обойтись одномерными структурами и получить $O(n\log^2 n)$ (–).

Разделяй и властвуй (CDQ). Первый ход — отсортировать по $a$ (индексы нам не нужны) (–). Далее функция solve(l, r) получает подотрезок, отсортированный по $a$, а возвращает отметки «убит» для пар внутри него и массив, отсортированный по $b$ (–, –):

  1. Делим пополам, рекурсивно решаем левую и правую половины. Все убийства внутри половин уже учтены (–). У правой половины $a$ больше, поэтому элементы правой половины могут убивать элементы левой, но не наоборот: нас интересуют убийства «из правой половины в левую» (–).
  2. Порядок внутри половин уже не важен для $a$, поэтому «пересортируем» по $b$: оба результата рекурсии отсортированы по $b$, остаётся слить их за линию (–). Для каждого элемента помним, из какой половины он пришёл.
  3. Идём по слитому массиву в порядке убывания $b$ (–). Элементы правой половины добавляем в структуру по $(c,d)$; для элементов левой половины делаем запрос: есть ли в структуре элемент с $c'>c$ и $d'>d$. Если есть, то у него $a$ больше (он из правой половины), $b$ больше (он уже добавлен), $c$ и $d$ больше — значит, нас можно выбросить (–, –).

Структура по $(c,d)$. Для запроса «существует ли $c'>c$ и $d'>d$» достаточно хранить «ступенчатый стек» из несравнимых по $(c,d)$ пар: $c_1d_2>\dots$ (–). Вставка пары $(C,D)$: lower_bound по $c$ находит первую запись с $c'\ge C$. Если у неё $d'\ge D$, то новая пара бесполезна — не добавляем. Иначе удаляем все предшествующие записи с $d'\le D$ (они теперь бесполезны) и вставляем $(C,D)$ (–). Запрос $(C,D)$: берём первую запись с $c'>C$; у неё $d'$ среди подходящих наибольшее, поэтому достаточно сравнить $d'>D$ (–). Лектор в споре замечает, что здесь вместо дерева отрезков годится set.

Оценка времени. Лектор называет итог $O(n\log n)$ («сортировка слиянием плюс $n\log n$»), но структура по $(c,d)$ даёт $O(\log)$ на элемент на каждом уровне рекурсии, то есть всего $O(n\log^2 n)$. Слияние вместо повторной сортировки экономит именно лишний логарифм сортировки — иначе было бы $\log^3$ (спор об этом — –).
struct Dominance4D {
    vector<array<int, 4>> P; vector<char> dead;
    Dominance4D(vector<array<int, 4>> p) : P(move(p)), dead(P.size(), 0) {   // a попарно различны
        sort(P.begin(), P.end());
        solve(0, P.size());
    }
    // возвращает индексы [l, r), отсортированные по b; дополнительно отмечает убитых «через границу» половин
    vector<int> solve(int l, int r) {
        if (r - l == 1) return {l};
        int m = (l + r) / 2;
        auto A = solve(l, m), B = solve(m, r);        // у B значения a больше, чем у A
        vector<int> mg; mg.reserve(r - l);
        size_t i = 0, j = 0;
        while (i < A.size() || j < B.size()) {        // слияние по b; при равных b правая половина идёт раньше
            if (j == B.size() || (i < A.size() && P[A[i]][1] < P[B[j]][1])) mg.push_back(A[i++]);
            else mg.push_back(B[j++]);
        }
        map<int, int> front;                          // «ступенчатый стек»: c растёт, d убывает
        for (int k = (int)mg.size() - 1; k >= 0; k--) {   // по убыванию b
            int id = mg[k], c = P[id][2], d = P[id][3];
            if (id >= m) {                            // из правой половины — добавляем в структуру
                auto it = front.lower_bound(c);
                if (it != front.end() && it->second >= d) continue;       // нас уже перекрывают
                if (it != front.end() && it->first == c) it = front.erase(it);
                while (it != front.begin() && prev(it)->second <= d) front.erase(prev(it));
                front[c] = d;
            } else {                                  // из левой — спрашиваем: есть ли c' > c и d' > d
                auto it = front.upper_bound(c);
                if (it != front.end() && it->second > d) dead[id] = 1;
            }
        }
        return mg;
    }
};

В группе удобно хранить минимум и максимум префиксных сумм (–).

Оговорка. В записи не показано, как границы групп обрезаются по области, где $a_t$ действительно максимум; без такого ограничения значение «сумма минус $a_t$» переоценивает ответ. Задача J. Пути по сетке через достопримечательности

Нужно посетить все достопримечательности, ходя по сетке; подсчитать число способов. Сведение, которое преподаватель уже рассказывала на прошлом занятии: вместо цикла из левого верхнего угла в правый нижний считаем пары непересекающихся путей — из клетки сразу за стартом в клетку сразу перед концом (–).

  • Состояние динамики $dp[i][j]$: один путь дошёл до достопримечательности $i$, другой — до $j$, и все достопримечательности вплоть до $\max(i,j)$ посещены (–, –).
  • Порядок. На каждом шаге сумма координат растёт, поэтому достопримечательности сортируем по $x+y$ — это порядок, в котором их можно посещать (–).
  • Непересечение. Если пути пересеклись, то хвосты можно обменять местами: получатся два других пути из тех же начал, но к переставленным концам; все достопримечательности по-прежнему посещены. Значит, лишние пары — это пары путей, пришедших в конец 2 и конец 1 вместо конца 1 и конца 2, и ответ — разность двух значений динамики (–).
  • Переходы. Пусть $i\ge j$. Если $i>j+1$, то достопримечательность $i-1$ уже посещена тем же путём, и берём $dp[i-1][j]$ умноженное на число путей из $i-1$ в $i$. Если $i=j+1$, то предыдущая достопримечательность этого пути — какая-то $k\le j$; перебираем $k$ и умножаем на число путей из $k$ в $i$ (–).

Переходы первого вида стоят $O(1)$ (всего $O(n^2)$), второго — $O(n)$, но таких состояний только $O(n)$, поэтому тоже $O(n^2)$ в сумме. Итог $O(n^2)$ плюс предпосчёт факториалов для числа путей между клетками (–).

Дерево отрезков в $2n$ ячейках

Когда пишут дерево отрезков «сверху», обычно выделяют массив на $4n$. На самом деле вершин не больше $2n$: на нижнем слое $n$ вершин, выше $n/2$, $n/4$ и так далее (–). Хочется индексацию, в которой нет пропусков (–). Стандартная нумерация — в порядке обхода в ширину (корень $0$, дальше $1,2,3,\dots$) оставляет «дыры», если дерево неполное (–).

Нумерация в порядке обхода в глубину. Пусть вершина получила номер $x$. Левому сыну даём $x+1$, а правому — $x+2\,(\text{размер левого отрезка})$: у левого поддерева на $s$ листьях ровно $2s-1$ вершин, так что правый сын идёт сразу за ним (–). Пропусков нет, и памяти нужно $2n$ вместо $4n$.

Есть и второй плюс: вершины лежат в памяти почти подряд, при спуске нет «странных прыжков» по памяти (–). Это особенно полезно, когда в вершине лежит что-то тяжёлое (строка, матрица переходов) — выигрыш по памяти и по кэшу примерно вдвое. Ведущий весь одиннадцатый класс писал так (–).

struct SegSum2n {
    int n; vector<ll> t;
    SegSum2n(const vector<ll>& a) : n(a.size()), t(2 * a.size()) { build(0, 0, n, a); }
    void build(int x, int l, int r, const vector<ll>& a) {           // отрезок [l, r)
        if (r - l == 1) { t[x] = a[l]; return; }
        int m = (l + r) / 2;
        int L = x + 1, R = x + 2 * (m - l);                          // у левого поддерева 2(m-l)-1 вершин
        build(L, l, m, a); build(R, m, r, a);
        t[x] = t[L] + t[R];
    }
    ll get(int x, int l, int r, int ql, int qr) {
        if (qr <= l || r <= ql) return 0;
        if (ql <= l && r <= qr) return t[x];
        int m = (l + r) / 2;
        return get(x + 1, l, m, ql, qr) + get(x + 2 * (m - l), m, r, ql, qr);
    }
    ll get(int l, int r) { return get(0, 0, n, l, r); }
};

Добавление, удаление, $k$-я порядковая статистика

Задача. Структура, которая умеет добавить число, удалить число и найти $k$-й по порядку элемент (–). Зал предлагает корневую, спуск по дереву отрезков, декартово дерево, ordered_set — ведущий отмечает, что последнее медленно и его нужно помнить (–).

Решение. Сожмём координаты (если запросы известны заранее) и заведём дерево отрезков (или Фенвика) по значениям: добавление числа $x$ — $+1$ в точке $x$, удаление — $-1$. $k$-й элемент — минимальный префикс с суммой не меньше $k$, находится спуском (–). Поскольку запросы только $+1$ и $-1$ в точке, достаточно дерева Фенвика (–).

  • Спуск по Фенвику можно заменить бинпоиском с префиксными суммами за $O(\log^2 n)$; ведущий считает, что это тоже быстро (константа около 100–400), и по факту часто быстрее дерева отрезков (–).
  • Если запросы приходят онлайн, делаем неявный Фенвик — массив заменяется на ассоциативный контейнер (–).
  • Сам спуск по Фенвику идёт так: в ячейке записана сумма на промежутке; если сумма в ячейке меньше $k$ — вычитаем её и сдвигаемся вправо, иначе уходим внутрь (–). Удобнее писать для нумерации с единицы.

Реализация спуска по Фенвику — метод kth в коде выше (раздел про Фенвика); он проверен в тесте против перебора. Вывод ведущего: «чем писать декартач — фигня»; такой способ намного проще.

Прибавление на отрезке и $\gcd$ на отрезке

Задача. Операции: $+= x$ на отрезке и $\gcd$ на отрезке (числа до $10^9$ или около того, $n\le 5\cdot 10^4$) (–). Кажется, что $\gcd$ непредсказуемо меняется при прибавлении (то получает новые делители, то теряет), и надо подумать (–). Зал и чат ищут идею; ключ — свойство алгоритма Евклида (–):

$\gcd(a,b)=\gcd(a-b,\,b)$

Следствие: $\gcd(a_l,a_{l+1},\dots,a_r)=\gcd(a_l,\ a_{l+1}-a_l,\ \dots,\ a_r-a_{r-1})$ — достаточно знать $a_l$ и разности соседних (–). Прибавление $x$ на $[l,r]$ меняет только две разности: $d_l\mathrel{+}=x$ и $d_{r+1}\mathrel{-}=x$ (–). Значит:

  • дерево отрезков по разностям поддерживает точечные изменения и $\gcd$ на отрезке;
  • значение $a_l$ — префиксная сумма разностей, хранится в дереве Фенвика (–).

Массовых операций (push) не нужно вовсе (–). Лектор уточняет, что догадаться до такого можно, если знать это свойство $\gcd$, а не «угадывая» сразу переход к разностям (–).

struct RangeAddGcd {
    int n, sz; vector<ll> d, seg; Fenwick bit;     // d[i] = a[i] - a[i-1]; bit хранит d, чтобы получить a[i]
    RangeAddGcd(const vector<ll>& a) : n(a.size()), d(a.size()), bit(a.size() + 1) {
        for (int i = 0; i < n; i++) { d[i] = a[i] - (i ? a[i - 1] : 0); bit.add(i + 1, d[i]); }
        sz = 1; while (sz < n) sz <<= 1;
        seg.assign(2 * sz, 0);
        for (int i = 1; i < n; i++) seg[sz + i] = llabs(d[i]);
        for (int v = sz - 1; v >= 1; v--) seg[v] = std::gcd(seg[2 * v], seg[2 * v + 1]);
    }
    void bump(int i, ll x) {
        if (i >= n) return;
        d[i] += x; bit.add(i + 1, x);
        if (i == 0) return;                           // d[0] = a[0] в дереве НОД не участвует
        int v = sz + i; seg[v] = llabs(d[i]);
        for (v >>= 1; v; v >>= 1) seg[v] = std::gcd(seg[2 * v], seg[2 * v + 1]);
    }
    void add(int l, int r, ll x) { bump(l, x); bump(r + 1, -x); }          // [l, r] += x
    ll gcd_query(int l, int r) {                                          // НОД на [l, r]
        ll g = llabs(bit.pref(l + 1));                // a_l
        ll res = 0;
        for (int lo = l + 1 + sz, hi = r + 1 + sz; lo < hi; lo >>= 1, hi >>= 1) {
            if (lo & 1) res = std::gcd(res, seg[lo++]);
            if (hi & 1) res = std::gcd(res, seg[--hi]);
        }
        return std::gcd(g, res);
    }
};

Наибольшая общая подпоследовательность двух перестановок

Задача. Даны две перестановки; найти длину их наибольшей общей подпоследовательности быстрее, чем за квадрат (–). Для обычных массивов за квадрат умеем, но у перестановок все элементы различны.

Сведение к НВП. Для каждого числа из первой перестановки запоминаем его позицию. Идём по второй перестановке и заменяем каждое число на его позицию в первой; получается последовательность позиций. Общая подпоследовательность — это те числа, которые стоят в одном порядке в обеих перестановках, то есть возрастающая подпоследовательность в новой последовательности (–). Ответ — длина НВП (–).

НВП через Фенвика. Динамика: $d_p$ — длина лучшей возрастающей цепочки, оканчивающейся на значении $p$. При приходе нового значения $p$ берём максимум $d$ на префиксе $[0,p-1]$ и записываем в ячейку $p$ значение «максимум $+1$» (–). Требуются «максимум на префиксе» и «присвоить в точке», причём новые значения в ячейках только возрастают — этого достаточно, чтобы вместо суммы в дереве Фенвика хранить максимум (–). Если бы значение в ячейке могло уменьшиться, пересчитать накопленные максимумы было бы нельзя.

int lcs_of_permutations(const vector<int>& p, const vector<int>& q) {   // значения 0..n-1
    int n = p.size();
    vector<int> pos(n);
    for (int i = 0; i < n; i++) pos[p[i]] = i + 1;
    FenwickMax f(n);
    int best = 0;
    for (int x : q) {                                 // идём по q, для каждого числа берём его позицию в p
        int len = f.get(pos[x] - 1) + 1;              // длина лучшей цепочки, кончающейся левее в p
        f.upd(pos[x], len);
        best = max(best, len);
    }
    return best;
}

Прибавление на отрезке и сумма на отрезке на Фенвике

Задача. $+= x$ на отрезке и сумма на отрезке — обычно решается деревом отрезков с push. Ведущий называет задачу «мозговзрывающей»: нужно обойтись деревом Фенвика (–). Если убрать любую из двух операций, задача легко решается разностным массивом; обе сразу — нет (–).

Пусть $d$ — разностный массив ($d_j=a_j-a_{j-1}$). Сумма на отрезке — разность двух префиксов (–). Сумма первых $r$ элементов:

$\displaystyle\sum_{i=1}^{r}a_i=\sum_{i=1}^{r}\sum_{j=1}^{i}d_j=\sum_{j=1}^{r}d_j\,(r-j+1)=(r+1)\sum_{j=1}^{r}d_j-\sum_{j=1}^{r}j\,d_j$

Элемент $d_j$ войдёт в сумму столько раз, сколько $i$ лежат на $[j,r]$ (–). Получаем две суммы по префиксу: одна по $d_j$, другая по $j\,d_j$ — два дерева Фенвика (–). Прибавление $x$ на $[l,r]$ — две точечные правки разностного массива в обоих деревьях (). 

// S(r) = sum_{j<=r} d_j (r - j + 1) = (r + 1) * sum d_j  -  sum j * d_j
struct RangeFenwick {
    Fenwick b1, b2;
    RangeFenwick(int n) : b1(n + 1), b2(n + 1) {}
    void upd(int i, ll x) { b1.add(i, x); b2.add(i, x * i); }
    void add(int l, int r, ll x) { upd(l, x); upd(r + 1, -x); }       // [l, r] += x, индексы с 1
    ll pref(int r) { return (r + 1) * b1.pref(r) - b2.pref(r); }
    ll get(int l, int r) { return pref(r) - pref(l - 1); }
};

Дерево отрезков без push

Ещё один способ решить ту же задачу. В массовых операциях обычно есть массив накопленных изменений (у нас — сколько нужно прибавить по поддереву), и функция push отправляет их вниз (–). Вызывать push на каждом шаге бывает дорого (в одной задаче с модулем и тяжёлой арифметикой это было причиной ТЛ) — можно не проталкивать изменения вниз (–, –).

  • В вершине $v$ храним $\mathrm{mod}_v$ — сколько нужно добавить всем элементам поддерева (изменения, которые на ней остановились) и $\mathrm{sum}_v$ — сумма на поддереве с учётом всех изменений внутри поддерева, но без тех, что лежат на пути к корню (–).
  • Обновление: разбиваем отрезок на накрытые вершины, добавляем $x$ в $\mathrm{mod}$ и $x\cdot\text{длина}$ в $\mathrm{sum}$; поднимаясь вверх, пересчитываем $\mathrm{sum}_v=\mathrm{sum}_{l}+\mathrm{sum}_{r}+\mathrm{mod}_v\cdot\text{длина}_v$ (–).
  • Запрос суммы: идём вниз как обычно; в каждой посещённой вершине добавляем $\mathrm{mod}_v\cdot(\text{пересечение отрезка вершины с запросом})$ — вклад модификатора, который не спускали вниз (–).

Отличие от обычного: с push изменение рано или поздно спускается вниз и «пропадает» из вершины, здесь оно остаётся в ней навсегда и учитывается каждый раз при пересчёте суммы предка и при запросе (–).

struct NoPushTree {
    int n; vector<ll> sum, mod;       // sum[v] учитывает все изменения внутри поддерева v; mod[v] — прибавка всем элементам v
    NoPushTree(const vector<ll>& a) : n(a.size()), sum(4 * a.size(), 0), mod(4 * a.size(), 0) { build(1, 0, n, a); }
    void build(int v, int l, int r, const vector<ll>& a) {
        if (r - l == 1) { sum[v] = a[l]; return; }
        int m = (l + r) / 2;
        build(2 * v, l, m, a); build(2 * v + 1, m, r, a);
        sum[v] = sum[2 * v] + sum[2 * v + 1];
    }
    void add(int v, int l, int r, int ql, int qr, ll x) {
        if (qr <= l || r <= ql) return;
        if (ql <= l && r <= qr) { mod[v] += x; sum[v] += x * (r - l); return; }
        int m = (l + r) / 2;
        add(2 * v, l, m, ql, qr, x); add(2 * v + 1, m, r, ql, qr, x);
        sum[v] = sum[2 * v] + sum[2 * v + 1] + mod[v] * (r - l);
    }
    ll get(int v, int l, int r, int ql, int qr) {
        if (qr <= l || r <= ql) return 0;
        if (ql <= l && r <= qr) return sum[v];
        int m = (l + r) / 2;
        ll res = mod[v] * (min(r, qr) - max(l, ql));      // вклад модификатора, который не спускали вниз
        return res + get(2 * v, l, m, ql, qr) + get(2 * v + 1, m, r, ql, qr);
    }
    void add(int l, int r, ll x) { add(1, 0, n, l, r, x); }
    ll get(int l, int r) { return get(1, 0, n, l, r); }
};
Когда работает. Приём подходит для операций, которые можно применять в любом порядке и сворачивать в одно значение: ведущие обсуждают, что нужны ассоциативность и коммутативность. Умножение матриц, где порядок запросов важен, так делать нельзя (–).

Позиции $i$, для которых $\operatorname{mex}(a[i..R])=\max(a[i..R])+1$

Задача. Дан массив неотрицательных чисел и $Q$ запросов-отрезков $[L,R]$ (все запросы известны заранее). Ответ на запрос: сколько позиций $i\in[L,R]$ таких, что на суффиксе $a[i..R]$ минимальное неотрицательное число, которого там нет ($\operatorname{mex}$), равно максимуму на этом суффиксе плюс один (–, –).

Переформулировка. Условие «$\operatorname{mex}=\max+1$» означает, что на суффиксе присутствуют все числа $0,1,\dots,\max$. Для неотрицательных чисел на суффиксе число различных значений всегда не больше $\max+1$ (возможные значения лежат в $[0,\max]$), а равенство достигается ровно когда присутствуют все (–, –). Поэтому условие равносильно $\max+1-\text{distinct}=0$ — а величина $\max+1-\text{distinct}$ неотрицательна, и «нулей» среди неотрицательных чисел можно считать как минимум и количество минимумов — ровно структура из задачи про прямоугольники (–). Ответ на запрос: возьмём минимум и количество минимумов на $[L,R]$; если минимум равен нулю — ответ равен количеству, иначе нулю (–).

Заметаем $R$ слева направо. Поддерживаем массив $v_i=\max(a[i..R])+1-\text{distinct}(a[i..R])$ при шаге $R\to R+1$ и отвечаем на все запросы с этой правой границей (–). Новый элемент $x$ меняет две вещи:

  • Число различных. Для суффиксов, начинающихся правее предыдущего вхождения $x$, число различных увеличивается на один; для остальных $x$ уже был. Значит, $v_i\mathrel{-}=1$ на $(\mathrm{prev}_x,\ R]$ (–).
  • Максимум. Храним стек суффиксных максимумов (блоки подряд идущих $i$, у которых $\max(a[i..R])$ одинаков). При добавлении $x$ пока верхний блок имеет максимум $\le x$, снимаем его и на его отрезке прибавляем $x-\text{старый максимум}$; новая позиция получает блок с максимумом $x$ (–). Каждый блок добавляется и удаляется один раз, поэтому всего $O(n)$ обновлений дерева.

Итого $O((n+Q)\log n)$. Ведущий для надёжности пересказывает идею второй раз (–).

    int n = a.size();
    MinCount tree(vector<ll>(n, 1));                // val[i] = max(a[i..R]) + 1 - (число различных в a[i..R])
    vector<vector<pair<int, int>>> byR(n);
    for (int i = 0; i < (int)qs.size(); i++) byR[qs[i].second].push_back({qs[i].first, i});
    vector<int> ans(qs.size(), 0);
    map<int, int> last;                             // последнее вхождение числа
    vector<array<int, 3>> st;                       // стек суффиксных максимумов: {начало, конец, значение}
    for (int R = 0; R < n; R++) {
        int x = a[R];
        auto it = last.find(x);
        int prevPos = it == last.end() ? -1 : it->second;
        last[x] = R;
        tree.update(prevPos + 1, R + 1, -1);        // +1 различное для всех левых границ правее прошлого вхождения
        int start = R;
        while (!st.empty() && st.back()[2] <= x) {  // поглощаем блоки, где максимум был не больше x
            auto [s, e, v] = st.back(); st.pop_back();
            tree.update(s, e + 1, x - v);           // старый максимум v заменяется на x
            start = s;
        }
        tree.update(R, R + 1, x + 1);               // новая позиция: максимум x и константа +1
        st.push_back({start, R, x});
        for (auto [L, id] : byR[R]) {
            auto [mn, c] = tree.query(L, R + 1);
            ans[id] = mn == 0 ? (int)c : 0;
        }
    }
    return ans;
}
Проверка. Для новой позиции в коде прибавляется $x+1$ (максимум плюс константа), а «$-1$ за новое различное» применяется ко всему промежутку $(\mathrm{prev}_x,R]$ — в сумме для позиции $R$ получается $x$. Ответы сверены с перебором $\operatorname{mex}$ и максимума на 500 случайных массивах.

Разбиения на отрезки со строго возрастающим OR

Задача. Дан массив из $n$ чисел. Сколько существует способов разбить его на подряд идущие отрезки так, чтобы побитовое OR на отрезках строго возрастало (–)? Нужно быстрее, чем за квадрат (–).

Ключевой факт. Если зафиксировать правый конец $i$ и двигать левый, то значение OR только растёт, причём каждый раз, когда оно меняется, в нём включается новый бит; значит, различных значений OR на отрезках с общим правым концом не больше $\log A$. Поэтому пар «правый конец — значение OR» в сумме $O(n\log A)$, хотя всех отрезков $O(n^2)$ (–, –).

Динамика «в лоб». $dp[i][x]$ — число разбиений префикса длины $i$, у которых последний отрезок имеет OR, равный $x$. Тогда $dp[i][x]=\sum_{j–). Проблемы: $x$ принимает до $A$ значений (память $nA$), и есть «индикатор» $\mathrm{OR}(j+1..i)=x$ (–).

  • Для данного $i$ значения $x$ — это не более $\log A$ различных OR, и каждому соответствует промежуток значений $j$ (OR монотонен), так что индикатор заменяется суммой по промежутку $j\in[l_x,r_x]$; для данного $i$ эти промежутки можно найти, поддерживая список групп при добавлении элемента (–).
  • Само $dp$ хранится только для значений $x$, реально встречающихся для данного префикса, то есть $O(n\log A)$ памяти (–).

Геометрия. Нарисуем каждое состояние $(j,y)$ как точку на плоскости с весом $dp[j][y]$. Тогда $dp[i][x]$ — сумма весов в прямоугольнике $j\in[l_x,r_x]$, $y–). Точек и запросов по $n\log A$ штук.

Заметающая прямая. Идём по $j$ слева направо; в дереве (по значению $y$) храним суммы уже добавленных точек. Прямоугольный запрос сумм — это разность двух префиксных по $j$: $F(r_x,x)-F(l_x-1,x)$, где $F(t,x)=\sum_{j\le t}\sum_{y–). К моменту, когда мы доходим до $i$, значения $dp[i][\cdot]$ уже посчитаны: порог запроса строго меньше $i$ (–). Значения $y$ можно сжать до чисел $0,1,\dots$ — тогда дерево на $O(n\log A)$ ячеек (–); без сжатия — неявное дерево (–). Время $O(n\log A\log n)$. Ответ — сумма $dp[n][\cdot]$ (–).

int or_partitions(const vector<int>& a) {
    int n = a.size();
    struct Grp { int x, lo, hi; };                  // OR(a[p..i]) = x для всех начал p в [lo, hi] (индексы с 1)
    vector<vector<Grp>> g(n + 1);
    vector<Grp> cur;
    for (int i = 1; i <= n; i++) {
        for (auto& e : cur) e.x |= a[i - 1];
        cur.push_back({a[i - 1], i, i});
        vector<Grp> nw;                             // склеиваем соседние группы с одинаковым OR; их не более log A
        for (auto& e : cur) {
            if (!nw.empty() && nw.back().x == e.x) nw.back().hi = e.hi; else nw.push_back(e);
        }
        cur = nw; g[i] = cur;
    }
    vector<int> xs = {-1};                          // -1 — виртуальное «предыдущее значение» для первого отрезка
    for (int i = 1; i <= n; i++) for (auto& e : g[i]) xs.push_back(e.x);
    sort(xs.begin(), xs.end()); xs.erase(unique(xs.begin(), xs.end()), xs.end());
    auto id = [&](int x) { return int(lower_bound(xs.begin(), xs.end(), x) - xs.begin()) + 1; };
    struct Qr { int i, k, sign; };                  // для dp[i][g[i][k].x] нужно прибавить sign * F(t, x)
    vector<vector<Qr>> at(n + 1);                   // at[t]: запросы с порогом t
    vector<vector<ll>> dp(n + 1);
    for (int i = 1; i <= n; i++) {
        dp[i].assign(g[i].size(), 0);
        for (int k = 0; k < (int)g[i].size(); k++) {
            at[g[i][k].hi - 1].push_back({i, k, +1});                       // j <= hi - 1
            if (g[i][k].lo - 2 >= 0) at[g[i][k].lo - 2].push_back({i, k, -1});  // j <= lo - 2
        }
    }
    Fenwick f(xs.size());                           // по значению OR храним сумму dp[j][y]
    for (int t = 0; t < n; t++) {
        if (t == 0) f.add(id(-1), 1);               // dp[0][-1] = 1
        else for (int k = 0; k < (int)g[t].size(); k++) f.add(id(g[t][k].x), dp[t][k] % MOD);
        for (auto& q : at[t]) {
            ll v = f.pref(id(g[q.i][q.k].x) - 1) % MOD;                      // сумма по y < x
            dp[q.i][q.k] = ((dp[q.i][q.k] + q.sign * v) % MOD + MOD) % MOD;
        }
    }
    ll ans = 0;
    for (ll v : dp[n]) ans = (ans + v) % MOD;
    return ans;
}
Проверка. Для первого отрезка добавлен виртуальный столбец $y=-1$ с единицей (чтобы любой OR подходил). Подсчёт сверен с перебором всех $2^{n-1}$ разбиений на 500 случайных массивах $n\le 14$.
