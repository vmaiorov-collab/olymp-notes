Задача E. Присвоение в вершине и сумма на пути

Дерево, запросы «присвоить число вершине» и «сумма на пути $v$–$u$». Строим эйлеров обход; присвоение превращаем в «$+=$ в точке» (старое значение запоминаем). Сумма на пути разбивается на три вертикальных пути: $v\to$ корень, $u\to$ корень, $\mathrm{lca}\to$ корень (–).

Вершина со значением $w$ вносит вклад в сумму до корня у всех вершин своего поддерева. Поэтому в Фенвике на позицию входа вершины добавляем $+w$, а на позицию выхода — $-w$; сумма на вертикальном пути — это префиксная сумма до позиции $v$ включительно (–).

Задача F. Сколько рёбер пути имеют вес не больше $x$

Сортируем запросы и рёбра по весу (события «ребро» и «запрос $x$»; при равенстве сначала рёбра, ведь знак нестрогий, –). Ребро, которое «включается», прибавляет $1$ ко всем вертикальным путям, проходящим через него: то есть $+1$ в точке входа нижнего конца и $-1$ в точке выхода, а путь $v$–$u$ снова разбивается на три вертикальных (–). Всего $O((n+q)\log n)$ с Фенвиком.

Задача G. «Новогодняя ёлка»: перекраска поддерева и число цветов

Корневое дерево, два запроса: перекрасить всё поддерево $v$ в цвет $c$ (цветов не больше $60$) и найти число различных цветов в поддереве. Эйлеров обход «сводит задачу на дереве к задаче на массиве с отрезками» (–): поддерево — отрезок $[tin_v,\ tout_v)$.

Дерево отрезков с присвоением на отрезке (массовая операция «присвоить $c$», в детях старые пометки перезатираются) и в каждой вершине — маска присутствующих цветов в одном 64-битном числе. Объединение — побитовое OR; ответ на запрос — OR масок $O(\log n)$ вершин и popcount (–). Цветов мало, поэтому «не нужно умножать на 60»: множество цветов помещается в одно слово (–).

struct ColorTree {
    int n; vector<ll> mask; vector<int> lz;
    ColorTree(const vector<int>& c) : n(c.size()), mask(4 * c.size(), 0), lz(4 * c.size(), -1) { build(1, 0, n, c); }
    void build(int v, int l, int r, const vector<int>& c) {
        if (r - l == 1) { mask[v] = 1LL << c[l]; return; }
        int m = (l + r) / 2; build(2 * v, l, m, c); build(2 * v + 1, m, r, c); mask[v] = mask[2 * v] | mask[2 * v + 1];
    }
    void apply(int v, int col) { mask[v] = 1LL << col; lz[v] = col; }
    void push(int v) { if (lz[v] >= 0) { apply(2 * v, lz[v]); apply(2 * v + 1, lz[v]); lz[v] = -1; } }
    void assign(int v, int l, int r, int ql, int qr, int col) {
        if (qr <= l || r <= ql) return;
        if (ql <= l && r <= qr) { apply(v, col); return; }
        push(v); int m = (l + r) / 2;
        assign(2 * v, l, m, ql, qr, col); assign(2 * v + 1, m, r, ql, qr, col);
        mask[v] = mask[2 * v] | mask[2 * v + 1];
    }
    ll get(int v, int l, int r, int ql, int qr) {
        if (qr <= l || r <= ql) return 0;
        if (ql <= l && r <= qr) return mask[v];
        push(v); int m = (l + r) / 2;
        return get(2 * v, l, m, ql, qr) | get(2 * v + 1, m, r, ql, qr);
    }
    void recolor(int l, int r, int col) { assign(1, 0, n, l, r, col); }
    int distinct(int l, int r) { return __builtin_popcountll(get(1, 0, n, l, r)); }
};

Задача «Вирусы»: пары предок–потомок в двух деревьях

Даны два дерева на одних и тех же вершинах. Нужно сосчитать пары $(v,u)$, где $v$ — предок $u$ и в первом, и во втором дереве (–).

Обходим второе дерево в глубину, а первое пронумеровываем эйлеровым обходом, так что поддерево $v$ в первом дереве — отрезок. В Фенвике по первому дереву храним «вершины, второй раз уже полностью обработанные». Когда заходим в $v$, запоминаем «красное» — сумму на отрезке поддерева $v$ первого дерева (в ней лишнее, пришедшее из прошлых ветвей). Когда выходим из $v$, берём ту же сумму и вычитаем красное: разность — это ровно вершины, добавленные, пока мы были внутри $v$, то есть потомки $v$ во втором дереве, лежащие в поддереве $v$ первого. После этого добавляем $+1$ в позицию самой $v$ (–). Итого $O(n\log n)$.

ll common_ancestor_pairs(const Tree& A, const Tree& Bt) {
    int n = A.n; Fen f(n); ll ans = 0;
    vector<int> before(n);
    vector<pair<int, int>> st{{0, 0}};
    before[0] = f.sum(A.tin[0], A.out[0]);
    while (!st.empty()) {
        int v = st.back().first, i = st.back().second;
        if (i < (int)Bt.adj[v].size()) {
            st.back().second++;
            int u = Bt.adj[v][i];
            if (u == Bt.par[v] && v != 0) continue;
            before[u] = f.sum(A.tin[u], A.out[u]);               // «красное»: что уже было в поддереве u первого дерева
            st.push_back({u, 0});
        } else {
            ans += f.sum(A.tin[v], A.out[v]) - before[v];        // «зелёное»: добавилось, пока мы были внутри v
            f.add(A.tin[v], 1);
            st.pop_back();
        }
    }
    return ans;
}
Проверка. $v\ne u$ учтено: $+1$ в позицию $v$ добавляется только после подсчёта разности. Сверено с перебором всех пар на 300 случайных пар деревьев.

Задача «Антивещество»: динамика с максимумами на отрезках

Ведущий замечает, что задачу (с Всероса 2016–2017) участники сдавали разреженной таблицей, и обещает рассказать решение без неё (–). Условие в записи пересказано неполно: есть $N$ экспериментов; $i$-й стоит $c_i$ и даёт случайное количество вещества из $[l_i,r_i]$; в любой момент вещества не больше $A$; в конце получаем $x\cdot 10^9-(\text{затраты})$, и нужно максимизировать гарантированную прибыль (–). Стратегия адаптивна: решение зависит от текущего количества вещества (–), поэтому состояние — только количество $x$.

Пусть $dp_x$ — минимальные гарантированные затраты, чтобы оказаться с $x$ веществом. Перебираем последний эксперимент $i$: до него было $y\in[x-r_i,\ x-l_i]$, худший случай — максимум затрат на этом отрезке: $$dp_x=\min_i\Bigl(c_i+\max_{y\in[x-r_i,\,x-l_i]} dp_y\Bigr),\qquad x-r_i\ge 0,\ x\le A,\quad dp_0=0$$ (–, –). Ответ — $\max_x\,(x\cdot 10^9-dp_x)$ (–). Левая часть окна неотрицательна, потому что $l_i>0$ и мы строго идём вправо: циклических зависимостей нет (–).

Остаётся быстро брать максимум на известных отрезках. Берётся «тот самый алгоритм», о котором ведущий рассказывал раньше: стек максимумов + СНМ, когда запросы заранее сгруппированы по правой границе (–). Идём слева направо; элемент, вытесненный из стека более крупным, «приклеивается» к вытеснившему в СНМ, а ответ на $[l,r]$ при правом конце $r$ — значение в корне $l$. Хранить сами запросы явно нельзя (их $N\cdot A$, «по памяти сдохнете»), поэтому их приходится порождать на лету (–); именно эта часть в записи объяснена сжато, и ведущий сам предлагает переспросить, если непонятно (–).

vector<ll> offline_range_max(const vector<ll>& a, const vector<pair<int, int>>& qs) {    // отрезки [l, r] включительно
    int n = a.size(); vector<vector<pair<int, int>>> byR(n);
    for (int i = 0; i < (int)qs.size(); i++) byR[qs[i].second].push_back({qs[i].first, i});
    vector<int> par(n), st; iota(par.begin(), par.end(), 0);
    function<int(int)> find = [&](int x) { while (par[x] != x) { par[x] = par[par[x]]; x = par[x]; } return x; };
    vector<ll> ans(qs.size());
    for (int i = 0; i < n; i++) {
        while (!st.empty() && a[st.back()] <= a[i]) { par[st.back()] = i; st.pop_back(); }   // вытесненный элемент «приклеивается» к вытеснившему
        st.push_back(i);
        for (auto [l, id] : byR[i]) ans[id] = a[find(l)];      // корень l — позиция максимума на [l, i]
    }
    return ans;
}
Что именно проверено. Приведён и протестирован общий приём «максимум на отрезках в оффлайне за $O((n+q)\,\alpha)$» (сверен с перебором). Саму игровую динамику выше целиком воспроизвести по записи нельзя: полное условие задачи не зачитывается, поэтому рекуррентность дана в том виде, в каком её вывел лектор.

Задача K. Сумма расстояний между равными значениями

В вершинах дерева записаны числа $a_i$; нужно найти $\sum_{i–). Равенство $a_i=a_j$ задаёт классы эквивалентности, поэтому перебирать пары не нужно: фиксируем значение $c$ и берём все вершины $v_1,\dots,v_{k_c}$ с этим значением (–).

На этих вершинах строим сжатое (виртуальное) дерево — оно содержит не более $2k_c$ вершин. Сумма попарных расстояний в дереве равна $\sum_{e} w_e\cdot s_e\,(k_c-s_e)$, где $s_e$ — число «наших» вершин под ребром $e$ (–). Построение — $O(k_c\log k_c)$, обход — $O(k_c)$; суммируя по значениям, получаем $O(n\log n)$ (–).

ll sum_pair_dist(const Tree& T, vector<int> v) {              // сумма расстояний по всем парам различных вершин
    sort(v.begin(), v.end()); v.erase(unique(v.begin(), v.end()), v.end());
    int k = v.size(); vector<int> nodes, pidx; build_virtual(T, v, nodes, pidx);
    vector<ll> cnt(nodes.size(), 0);
    for (int x : v) cnt[lower_bound(nodes.begin(), nodes.end(), x, [&](int a, int b) { return T.tin[a] < T.tin[b]; }) - nodes.begin()] = 1;
    ll ans = 0;
    for (int i = nodes.size() - 1; i >= 1; i--) {                // дети всегда имеют больший tin, чем родитель
        ans += (ll)(T.dep[nodes[i]] - T.dep[nodes[pidx[i]]]) * cnt[i] * (k - cnt[i]);
        cnt[pidx[i]] += cnt[i];
    }
    return ans;
}
Проверка. Виртуальное дерево строится по вершинам, отсортированным по времени входа, добавлением LCA соседних пар; родитель в сжатом дереве — LCA предыдущей и текущей вершин. Сумма расстояний сверена с перебором на 300 деревьях.

Задача «Туризм»: минимальное число вершин для набора достопримечательностей

Есть дерево с достопримечательностями в вершинах; путешественники разбивают их на отрезки подряд идущих. Для путешественника с отрезком $[L,R]$ нужно найти, сколько различных вершин он обязан посетить (путь начинается и кончается где угодно) (–).

Берём вершины, соответствующие достопримечательностям из $[L,R]$ (их не больше $R-L+1$, повторы не мешают), строим на них сжатое дерево. Стартовать нужно с какой-то вершины сжатого дерева (иначе лишний хвост выкинем), и мы обязаны побывать во всех его вершинах. Число вершин — сумма длин (в исходных рёбрах) всех сжатых рёбер плюс один (–). Отрезки не пересекаются, поэтому всё вместе работает за $O(n\log n)$.

int steiner_size(const Tree& T, const vector<int>& v) {      // сколько вершин в наименьшем поддереве, содержащем все v
    vector<int> nodes, pidx; build_virtual(T, v, nodes, pidx);
    int r = 1;
    for (size_t i = 1; i < nodes.size(); i++) r += T.dep[nodes[i]] - T.dep[nodes[pidx[i]]];
    return r;
}

Изменение элемента и сумма на отрезке

Массив из $n$ чисел, запросы: присвоить $a_i=x$ и найти сумму на отрезке. Дерево отрезков делает и то и другое за $O(\log n)$, но можно сделать присвоение за $O(1)$, а сумму — за $O(\sqrt n)$ (–). Лектор рассказывает, что идею «корнячки» можно придумать самому и что Денис Павлович Кириенко придумал её сам на олимпиаде 90-х (–).

Идея. Разбиваем массив на блоки по $B$ элементов и для каждого блока храним сумму. Изменение элемента: поправить $a_i$ и сумму блока $\lfloor i/B\rfloor$ — $O(1)$ (–). Запрос $[l,r)$: часть блоков лежит целиком — их суммы добавляем сразу; не более двух блоков задеты частично — у одного суффикс запроса, у другого префикс; их проходим по одному элементу (–).

Выбор $B$. Запрос стоит $O(B)$ (края) плюс $O(n/B)$ (число блоков). Сумма $B+n/B$ минимальна, когда слагаемые равны: $B=n/B$, то есть $B=\sqrt n$ (–). Для ноль-индексации номер блока элемента $i$ — это $\lfloor i/B\rfloor$ (–).

Лаконичная реализация. Идём по $i$ от $l$ до $r$: если $i$ — начало блока и блок целиком помещается ($i+B\le r$), добавляем сумму блока и делаем $i\mathrel{+}=B$; иначе добавляем $a_i$ и делаем $i\mathrel{+}=1$ (–). Операция взятия остатка там много раз, но почти всегда это приемлемо (–).

struct BlockSum {
    int n, B; vector<ll> a, blk;
    BlockSum(const vector<ll>& v) : n(v.size()), B(isqrt_ceil(v.size())), a(v), blk((v.size() + B - 1) / B + 1, 0) {
        for (int i = 0; i < n; i++) blk[i / B] += a[i];
    }
    void assign(int i, ll x) { blk[i / B] += x - a[i]; a[i] = x; }          // O(1)
    ll sum(int l, int r) {                                                  // O(sqrt n)
        ll s = 0;
        for (int i = l; i < r;) {
            if (i % B == 0 && i + B <= r) { s += blk[i / B]; i += B; }     // перепрыгнули через целый блок
            else { s += a[i]; i++; }                                        // идём по префиксу/суффиксу по одному
        }
        return s;
    }
};
Проверка. Реализация использует полуинтервал $[l,r)$ и размер блока $\lceil\sqrt n\rceil$; последний блок может быть короче — он никогда не берётся целиком, потому что $i+B>n\ge r$. Сверено с перебором на 300 массивах.

Запрос за $O(1)$, изменение за $O(\sqrt n)$

Теперь наоборот: запросов суммы много, изменений мало. Нужно хранить префиксные суммы и научиться чинить их за $O(\sqrt n)$ (–). Изменение $a_i$ меняет все префиксы справа от $i$ на одну и ту же разность. Поэтому делим префиксы на два слоя: в блоке $i$ перестраиваем префиксы внутри блока честно ($\le B$ операций); всем блокам правее заводим «отложенную» добавку к сумме всех предыдущих блоков ($\le n/B$ операций) (–). Префикс до $i$ — это префикс внутри блока плюс сумма блоков до него, поэтому запрос — $O(1)$.

struct PrefixBlocks {
    int n, B; vector<ll> a, in, before;          // in[i] — сумма a от начала блока до i включительно; before[b] — сумма всех блоков до b
    PrefixBlocks(const vector<ll>& v) : n(v.size()), B(isqrt_ceil(v.size())), a(v), in(v.size()), before((v.size() + B - 1) / B + 1, 0) {
        int nb = (n + B - 1) / B;
        for (int b = 0; b < nb; b++) {
            ll run = 0;
            for (int i = b * B; i < min(n, (b + 1) * B); i++) { run += a[i]; in[i] = run; }
            before[b + 1] = before[b] + run;
        }
    }
    void add(int i, ll d) {
        a[i] += d;
        for (int j = i; j < min(n, (i / B + 1) * B); j++) in[j] += d;      // свой блок пересчитываем честно
        for (int b = i / B + 1; b < (int)before.size(); b++) before[b] += d;  // остальным блокам — «отложенная» добавка
    }
    ll pref(int i) { return i == 0 ? 0 : in[i - 1] + before[(i - 1) / B]; }   // сумма a[0..i)
    ll sum(int l, int r) { return pref(r) - pref(l); }
};
Проверка. Обе структуры (суммы блоков и префиксы) сверены между собой и с перебором при случайных присвоениях и запросах; при $B=\lceil\sqrt n\rceil$ обе операции укладываются в $O(\sqrt n)$ (или $O(1)$ для запроса префиксной структуры).

Прибавление на отрезке и «сколько элементов не больше $x$»

Операции: $+x$ на отрезке и «сколько $a_i\le x$ на отрезке»; желаемая оценка — $O(n+q\sqrt n\ \cdot\text{что-то малое})$ вместо $O(n\log n)$ (–).

Сначала решаем задачу без прибавления. Блоки храним отсортированными (рядом лежит и исходный массив). Запрос: в целых блоках делаем бинарный поиск, в краях просматриваем элементы по одному. Время запроса $O(B+\frac nB\log B)$ (–).

Прибавление. Для целиковых блоков прибавляем к метке блока «ко всем элементам блока нужно добавить $x$»; в неполных (их не более двух) изменяем элементы и перестраиваем сортировку блока ($B\log B$). Тогда истинное значение — $a_i+\mathrm{lazy}_{\text{блока}}$, а бинарный поиск ведём по $x-\mathrm{lazy}$ (–).

struct SortedBlocks {
    int n, B; vector<ll> a, lazy; vector<vector<ll>> s;    // истинное значение = a[i] + lazy[блок]
    SortedBlocks(const vector<ll>& v, int B) : n(v.size()), B(B), a(v), lazy((v.size() + B - 1) / B, 0), s(lazy.size()) {
        for (int b = 0; b < (int)s.size(); b++) rebuild(b);
    }
    void rebuild(int b) { s[b].assign(a.begin() + b * B, a.begin() + min(n, (b + 1) * B)); sort(s[b].begin(), s[b].end()); }
    void part(int b, int l, int r, ll x) { for (int i = l; i < r; i++) a[i] += x; rebuild(b); }
    void add(int l, int r, ll x) {                         // [l, r)
        if (l >= r) return;
        int bl = l / B, br = (r - 1) / B;
        if (bl == br) { part(bl, l, r, x); return; }
        part(bl, l, (bl + 1) * B, x); part(br, br * B, r, x);
        for (int b = bl + 1; b < br; b++) lazy[b] += x;
    }
    int count_le(int l, int r, ll x) {                     // сколько a[i] <= x на [l, r)
        if (l >= r) return 0;
        int bl = l / B, br = (r - 1) / B, res = 0;
        auto scan = [&](int lo, int hi) { for (int i = lo; i < hi; i++) res += a[i] + lazy[i / B] <= x; };
        if (bl == br) { scan(l, r); return res; }
        scan(l, (bl + 1) * B); scan(br * B, r);
        for (int b = bl + 1; b < br; b++) res += upper_bound(s[b].begin(), s[b].end(), x - lazy[b]) - s[b].begin();
        return res;
    }
};
Уточнение про время. Лектор записывает цену изменения как $B\log B+n/B$ (–). Это верно, если перестраивать блок сортировкой; но изменённая часть блока и неизменённая каждая остаются отсортированными, так что их можно слить за $O(B)$, и цена изменения становится $O(B+n/B)$. Реализация ниже сортирует блок заново (проще), на асимптотику $\sqrt{n\log n}$ влияет только логарифм.

Какой $B$ лучше? Если взять $B=\sqrt n$, получим $O(n\sqrt n\log)$. Но запрос — $B+\frac nB\log B$, и слагаемые равны при $B^2=n\log n$ (используется $\log B\approx\log n$). Тогда $B=\sqrt{n\log n}$ и запрос работает за $O(\sqrt{n\log n})$, т.е. логарифм «уехал под корень» (–). Общий вывод: если работа на целый блок нетривиальна (бинарный поиск и т.п.), блок стоит брать больше $\sqrt n$ (–).

Эту задачу до корневой не умели решать никак: «$+=$ на отрезке» ломает деревья сортировок (–).

Проверка. Размер блока в тесте выбирался случайным (от $1$ до $8$), чтобы проверить и короткие, и длинные блоки; ответы сверены с перебором на 300 массивах.

Конструктив: обойти $n$ точек быстрее, чем за $n^2$

Отход от корнячки к конструктиву: на плоскости $n$ точек с координатами до $n$; стоим в $(0,0)$, за шаг двигаемся на $1$ вверх/вниз/влево/вправо. Нужно обойти все точки асимптотически быстрее $n^2$ (–). Подход «граф и кратчайший цикл» не нужен — минимум не требуется, нужен лишь путь, короче квадрата (–). Обход всей сетки даёт $n^2$ (–), пропуск пустых столбцов ломается на контрпримере (–).

Хорошая идея из зала — разбить плоскость на квадраты и обойти их; но в квадрате остаётся та же задача. Лектор разбирает: квадраты со стороной $\sqrt n$ можно пройти змейкой за $O(n)$ шагов между квадратами, а вот что делать внутри — неясно (–). Идею «тяжёлые/лёгкие квадраты» и «сортировать внутри» лектор доводит до решения, меняя только форму кусков:

Делим плоскость на полосы шириной $B$ (вторая сторона — $n$). Точки сортируем по возрастанию $y$ и обходим по полосам. Время складывается из двух частей: движение по $y$ — не более $n$ на полосу, то есть $\frac nB\cdot n$; движение по $x$ — когда переходим от прошлой точки в той же полосе, сдвиг не более $B$, то есть $n\cdot B$ в сумме (первые точки полос дают ещё $\frac nB\cdot n$) (–). Итого $O\!\left(\frac{n^2}{B}+nB\right)$, при $B=\sqrt n$ получается $O(n\sqrt n)$ (–).

vector<int> strip_order(const vector<pair<int, int>>& p, int B, bool snake) {
    vector<int> id(p.size()); iota(id.begin(), id.end(), 0);
    sort(id.begin(), id.end(), [&](int i, int j) {
        int bi = p[i].first / B, bj = p[j].first / B;
        if (bi != bj) return bi < bj;
        if (snake && bi % 2) return p[i].second > p[j].second;     // на нечётных полосах идём вниз
        return p[i].second < p[j].second;
    });
    return id;
}

Поиск mex на отрезке и алгоритм Мо

Теперь задача: массив неотрицательных чисел, запросы $[l,r]$ — найти $\operatorname{mex}$ значений на отрезке. Дереву отрезков непонятно, что хранить в вершине; в зале предлагают оффлайн, персистентность и последнюю позицию значения — всё это работает на массиве, но потом понадобятся модификации (мекс на пути, изменения), с которыми так не получится (–).

Решаем вспомогательную задачу: множество с операциями «добавить», «удалить», «найти mex» — для него хватает set недостающих значений и счётчика вхождений, всё за $O(\log n)$ (–). Ключевое наблюдение: если известен ответ для отрезка, то для соседнего его можно получить, удалив и добавив несколько элементов (–). Остаётся найти порядок запросов, при котором таких удалений и добавлений мало (–).

Каждый запрос $[l,r]$ — точка плоскости с координатами $(l,r)$; «сдвинуть границу на единицу» — шаг по плоскости. Мы свели задачу к обходу точек на плоскости предыдущим приёмом (–). Это и есть алгоритм Мо.

Реализация. Сортируем запросы компаратором: если $l/B$ равны, то по $r$; иначе по $l$ (–). Текущий отрезок $[L,R)$ изначально пуст (–). Чтобы не удалять несуществующие элементы, сначала расширяем границы, потом сужаем: четыре цикла (–).

Змейка. Оптимизация: после первого блока не возвращаться в начало, а на нечётных блоках идти по $r$ в обратную сторону. Достаточно в компараторе для нечётного блока $l/B$ поменять знак сравнения (–).

vector<int> mo_order(const vector<int>& L, const vector<int>& R, int B, bool snake = true) {
    vector<int> id(L.size()); iota(id.begin(), id.end(), 0);
    sort(id.begin(), id.end(), [&](int i, int j) {
        int bi = L[i] / B, bj = L[j] / B;
        if (bi != bj) return bi < bj;
        return (snake && bi % 2) ? R[i] > R[j] : R[i] < R[j];
    });
    return id;
}

Асимптотика. Двигаем границы $O(n\sqrt q)$ раз (при хорошем $B$), и каждое движение стоит $O(\log n)$ с set. Но запросов самого mex всего $q$, а сдвигов $n\sqrt n$, т.е. нужно научиться делать добавление и удаление за $O(1)$ ценой запроса $O(\sqrt n)$ (–).

Структура для mex: добавление за $O(1)$, запрос за $O(\sqrt n)$

Элементы больше $n$ можно сразу игнорировать: среди $n$ значений mex не превосходит $n$ (–). Храним массив cnt[v] — сколько раз значение встречается — и делим его на блоки; в каждом блоке храним число нулей (–). Изменение: при $+1$, если было $0$, уменьшаем счётчик нулей блока; при $-1$, если стало $0$, увеличиваем (–). Запрос: идём по блокам, пропуская те, где нулей нет, и в первом подходящем блоке ищем ноль честно (–). Итого $O(\sqrt n)$ на запрос и $O(1)$ на изменение.

struct MexSet {
    int V, B; vector<int> cnt, zeros;               // zeros[b] — сколько значений в блоке b отсутствуют
    MexSet(int V) : V(V), B(isqrt_ceil(V)), cnt(V, 0), zeros((V + B - 1) / B, 0) { for (int v = 0; v < V; v++) zeros[v / B]++; }
    void add(int x) { if (x < V && cnt[x]++ == 0) zeros[x / B]--; }
    void rem(int x) { if (x < V && --cnt[x] == 0) zeros[x / B]++; }
    int mex() const {
        int b = 0;
        while (b < (int)zeros.size() && zeros[b] == 0) b++;      // первый блок, где есть «дырка»
        if (b == (int)zeros.size()) return V;
        int v = b * B; while (cnt[v]) v++;                       // внутри блока ищем честно
        return v;
    }
};
// mex на отрезках [l, r] (включительно) алгоритмом Мо
vector<int> mo_mex(const vector<int>& a, const vector<pair<int, int>>& qs) {
    int n = a.size(), q = qs.size();
    vector<int> L(q), R(q);
    for (int i = 0; i < q; i++) { L[i] = qs[i].first; R[i] = qs[i].second; }
    MexSet ms(n + 1);
    vector<int> ans(q), ord = mo_order(L, R, isqrt_ceil(n));
    int l = 0, r = -1;                                           // текущий отрезок [l, r], пока пустой
    for (int i : ord) {
        while (l > L[i]) ms.add(a[--l]);                         // сначала расширяем...
        while (r < R[i]) ms.add(a[++r]);
        while (l < L[i]) ms.rem(a[l++]);                         // ...потом сужаем
        while (r > R[i]) ms.rem(a[r--]);
        ans[i] = ms.mex();
    }
    return ans;
}

Так mex на отрезке решается за $O(n\sqrt q+q\sqrt n)$ — «благодаря тому, что внутри Мо применили корнячку» (–). Перед этим ведущие делали перерыв (–; в записи там пауза и проверка звука).

Проверка. Мо с «змейкой» и структурой MexSet сверен с перебором на 300 массивах (границы включительные, значения вплоть до $n$ и выше).

Мо на дереве: mex на пути

Те же значения, но теперь они в вершинах дерева, а запросы — на путях (–). Нужен эйлеров обход; чтобы получился отрезок, записываем вершину при входе и при выходе, т.е. каждую вершину два раза (–). Для пути $u\to v$ рассмотрим отрезок обхода между ними: вершины, лежащие на пути, встречаются в нём ровно один раз, все остальные — два (–).

Поэтому Мо работает почти без изменений: у каждой вершины заводим счётчик «сколько раз она в отрезке»; на первом вхождении добавляем её значение в структуру, на втором — удаляем (то есть переключаем) (–, –). Для отрезка берём $[\mathrm{tin}_u,\ \mathrm{tin}_v]$, если $u$ — предок $v$, и $[\mathrm{tout}_u,\ \mathrm{tin}_v]$ иначе; во втором случае LCA в отрезок не попадает, его значение нужно добавить в структуру отдельно на время запроса (–, –). Если значения записаны на рёбрах, записываем значение при первом и последнем проходе по ребру — тогда LCA добавлять не нужно (–).

// mex значений на путях дерева: Мо по эйлерову обходу длины 2n, вершины, встретившиеся дважды, «гасят» друг друга
vector<int> tree_mex(const Tree& T, const vector<int>& val, const vector<pair<int, int>>& qs) {
    int n = T.n, q = qs.size();
    vector<int> L(q), R(q), extra(q, -1);
    for (int i = 0; i < q; i++) {
        int u = qs[i].first, v = qs[i].second;
        if (T.e_in[u] > T.e_in[v]) swap(u, v);
        int w = T.lca(u, v);
        if (w == u) { L[i] = T.e_in[u]; R[i] = T.e_in[v]; }
        else { L[i] = T.e_out[u]; R[i] = T.e_in[v]; extra[i] = w; }   // LCA в отрезок не попадает — добавим отдельно
    }
    MexSet ms(n + 1); vector<char> in(n, 0);
    auto flip = [&](int pos) { int v = T.ord2[pos]; if (in[v] ^= 1) ms.add(val[v]); else ms.rem(val[v]); };
    vector<int> ans(q), ord = mo_order(L, R, isqrt_ceil(2 * n));
    int l = 0, r = -1;
    for (int i : ord) {
        while (l > L[i]) flip(--l);
        while (r < R[i]) flip(++r);
        while (l < L[i]) flip(l++);
        while (r > R[i]) flip(r--);
        if (extra[i] >= 0) ms.add(val[extra[i]]);
        ans[i] = ms.mex();
        if (extra[i] >= 0) ms.rem(val[extra[i]]);
    }
    return ans;
}
Проверка. Дерево обходится итеративно (без рекурсии), значение LCA добавляется на время ответа и снимается. Сверено с перебором путей (ответ на путь по предкам через LCA) на 300 деревьях до 25 вершин.

3D-Мо: mex на отрезке с изменениями

Теперь к запросам mex добавляются изменения: «поставить в позицию $i$ новое значение». Раньше Мо опирался на то, что запросы известны заранее и нет изменений (–). Лектор признаётся, что Мо на олимпиадах почти не писал: однажды ему пришлось писать «Мо на дереве» на раунде Codeforces, и эту задачу он вспоминает с гордостью (–).

Запрос характеризуется тремя числами: $(l,\ r,\ t)$, где $t$ — сколько изменений применено до него. Состояние — точка в кубе. Сдвигать $l$ и $r$ мы умеем; сдвиг $t\to t+1$ тоже простой: если изменяемая позиция лежит внутри $[l,r]$, то удаляем старое значение и добавляем новое, иначе просто меняем массив; $t\to t-1$ — обратная операция (–).

Осталось выбрать порядок. Сортируем по тройке $(\lfloor l/B\rfloor,\ \lfloor r/B\rfloor,\ t)$ (–). Оценка: пока блоки $l$ и $r$ не меняются, $t$ растёт монотонно — не более $n$ шагов на пару блоков, а таких пар $(n/B)^2$, то есть $\frac{n^3}{B^2}$; границы $l$ внутри блока двигаются на $B$ за запрос — $nB$; $r$ в пределах блока — $nB$, а при смене блока $r$ — $\frac{n^2}{B}$ в сумме (–): $$O\!\left(nB+\frac{n^2}{B}+\frac{n^3}{B^2}\right).$$ При $B=n^{2/3}$ получаем $O(n^{5/3})$ (–). Это и называется 3D-Мо (–).

vector<int> mo_mex_updates(vector<int> a, const vector<Upd>& upd, const vector<array<int, 3>>& qs /* l, r, сколько изменений уже применено */) {
    int n = a.size(), q = qs.size();
    vector<int> oldv(upd.size()); { vector<int> b = a; for (size_t k = 0; k < upd.size(); k++) { oldv[k] = b[upd[k].pos]; b[upd[k].pos] = upd[k].val; } }
    int B = max(1, (int)cbrt((double)n * n));                      // блок ~ n^(2/3)
    vector<int> id(q); iota(id.begin(), id.end(), 0);
    sort(id.begin(), id.end(), [&](int i, int j) {
        const auto &x = qs[i], &y = qs[j];
        if (x[0] / B != y[0] / B) return x[0] / B < y[0] / B;
        if (x[1] / B != y[1] / B) return x[1] / B < y[1] / B;
        return x[2] < y[2];
    });
    MexSet ms(n + 1); int l = 0, r = -1, t = 0; vector<int> ans(q);
    auto apply = [&](int k, bool forward) {
        int p = upd[k].pos, nv = forward ? upd[k].val : oldv[k], ov = forward ? oldv[k] : upd[k].val;
        if (l <= p && p <= r) { ms.rem(ov); ms.add(nv); }
        a[p] = nv;
    };
    for (int i : id) {
        int L = qs[i][0], R = qs[i][1], T = qs[i][2];
        while (l > L) ms.add(a[--l]);
        while (r < R) ms.add(a[++r]);
        while (l < L) ms.rem(a[l++]);
        while (r > R) ms.rem(a[r--]);
        while (t < T) apply(t++, true);
        while (t > T) apply(--t, false);
        ans[i] = ms.mex();
    }
    return ans;
}
Проверка. Старые значения для отката изменений вычисляются заранее прогоном по списку изменений. Сверено с перебором (с применением первых $t$ изменений) на 300 примерах; блок взят как $\lfloor n^{2/3}\rfloor$.

Сумма разностей индексов равных элементов

Для запроса $[l,r]$ нужно найти $\sum_{x–), и снова предлагает посмотреть «в стиле Мо»: достаточно уметь расширять и сужать отрезок на один элемент (–).

Пусть справа добавляется элемент в позиции $R$, а равные ему сейчас стоят в позициях $i_1,i_2,i_3,\dots$ внутри отрезка. Тогда ответ растёт на $(R-i_1)+(R-i_2)+\dots=\mathrm{cnt}\cdot R-\sum i_j$. Нужно хранить для каждого значения только количество вхождений и сумму их индексов (–). Добавление слева — симметрично: прибавляется $\sum i_j-\mathrm{cnt}\cdot L$; удаление — те же формулы с вычитанием (–, –). Значения сжимаем или храним в хеш-таблице (–).

vector<ll> mo_equal_pairs(vector<int> a, const vector<pair<int, int>>& qs) {
    int n = a.size(), q = qs.size();
    { vector<int> s = a; sort(s.begin(), s.end()); s.erase(unique(s.begin(), s.end()), s.end());
      for (int& x : a) x = lower_bound(s.begin(), s.end(), x) - s.begin(); }
    vector<ll> cnt(n, 0), sumIdx(n, 0), res(q); ll cur = 0;
    auto add_right = [&](int p) { int v = a[p]; cur += cnt[v] * p - sumIdx[v]; cnt[v]++; sumIdx[v] += p; };   // новый индекс — самый большой
    auto add_left  = [&](int p) { int v = a[p]; cur += sumIdx[v] - cnt[v] * p; cnt[v]++; sumIdx[v] += p; };   // новый индекс — самый маленький
    auto rem_right = [&](int p) { int v = a[p]; cnt[v]--; sumIdx[v] -= p; cur -= cnt[v] * p - sumIdx[v]; };
    auto rem_left  = [&](int p) { int v = a[p]; cnt[v]--; sumIdx[v] -= p; cur -= sumIdx[v] - cnt[v] * p; };
    vector<int> L(q), R(q);
    for (int i = 0; i < q; i++) { L[i] = qs[i].first; R[i] = qs[i].second; }
    int l = 0, r = -1;
    for (int i : mo_order(L, R, isqrt_ceil(n))) {
        while (l > L[i]) add_left(--l);
        while (r < R[i]) add_right(++r);
        while (l < L[i]) rem_left(l++);
        while (r > R[i]) rem_right(r--);
        res[i] = cur;
    }
    return res;
}

Корневая оптимизация и корневая по запросам

Дальше ведёт Денис, затем (с ) блок про корневую по запросам рассказывает второй ведущий. Общая идея первой части: если у объектов выполнено какое-то суммарное ограничение, то «больших» объектов мало, и с ними можно обращаться иначе, чем с «маленькими».

Главная идея: сумма равна $S$ ⇒ больших не больше $\sqrt S$

Пусть $a_1+\dots+a_k=S$. Разобьём элементы на «не больше $\sqrt S$» и «больше $\sqrt S$». Если каждый элемент второй группы больше $\sqrt S$, а в сумме они дают не более $S$, то их не больше $\sqrt S$ (–). Константа $B$ вместо точного корня не важна: она одинаково входит в обе части оценки (–).

Число треугольников в графе

Неориентированный граф на $n$ вершинах и $m$ рёбрах; нужно число треугольников (–). Сумма степеней равна $2m$, поэтому вершин со степенью больше $B$ («тяжёлых») не больше $2m/B$; остальные — «лёгкие» (–).

Треугольники, где есть хотя бы одна лёгкая вершина. Перебираем вершину $v$ и по её списку смежности отмечаем её соседей (это «строка матрицы смежности», строящаяся на ходу). Перебираем лёгкого соседа $u$ и затем его соседей $w$ — их не больше $B$ — и проверяем, помечена ли $w$ (–). Суммарно $O(m+mB)=O(mB)$, потому что каждое ребро $vu$ рассматривается дважды (–).

Один треугольник найдётся несколько раз (зависит от числа лёгких вершин в нём), поэтому считаем $c_1,c_2,c_3$ — по числу лёгких вершин — и делим на соответствующие кратности (–). Треугольники из одних тяжёлых. Тяжёлых не больше $2m/B$; нумеруем их и строим матрицу смежности $\frac{2m}B\times\frac{2m}B$. Перебор всех троек — $O\!\left((m/B)^3\right)$, при $B=\sqrt m$ это $O(m\sqrt m)$ (–).

// (a) делим вершины на лёгкие (deg <= B) и тяжёлые; тяжёлых не больше 2m / B
ll triangles_heavy_light(int n, const vector<pair<int, int>>& edges) {
    int m = edges.size(), B = isqrt_ceil(2LL * m);
    vector<vector<int>> adj(n);
    for (auto [u, v] : edges) { adj[u].push_back(v); adj[v].push_back(u); }
    vector<char> heavy(n); for (int v = 0; v < n; v++) heavy[v] = (int)adj[v].size() > B;
    ll c[4] = {0, 0, 0, 0}; vector<char> mark(n, 0);
    for (int v = 0; v < n; v++) {
        for (int x : adj[v]) mark[x] = 1;                        // строка матрицы смежности для v
        for (int u : adj[v]) if (!heavy[u])                      // лёгкий сосед u: у него не больше B соседей
            for (int w : adj[u]) if (mark[w]) c[!heavy[v] + !heavy[u] + !heavy[w]]++;
        for (int x : adj[v]) mark[x] = 0;
    }
    // тройка (v; лёгкий u; w) встречается 2k раз у треугольника с k лёгкими вершинами
    ll ans = c[1] / 2 + c[2] / 4 + c[3] / 6;
    vector<int> hv, hid(n, -1);
    for (int v = 0; v < n; v++) if (heavy[v]) { hid[v] = hv.size(); hv.push_back(v); }
    int h = hv.size(); vector<vector<char>> g(h, vector<char>(h, 0));
    for (auto [u, v] : edges) if (heavy[u] && heavy[v]) g[hid[u]][hid[v]] = g[hid[v]][hid[u]] = 1;
    for (int i = 0; i < h; i++) for (int j = i + 1; j < h; j++) if (g[i][j])
        for (int k = j + 1; k < h; k++) ans += g[i][k] && g[j][k];
    return ans;
}
Кратности. Лектор говорит «$c_1$ считается один раз, $c_3$ — три раза» (–), а затем «шесть». В реализации выше перебираются упорядоченные тройки $(v;\ \text{лёгкая } u;\ w)$, и треугольник с $k$ лёгкими вершинами находится ровно $2k$ раз; поэтому делим $c_1$ на $2$, $c_2$ на $4$, $c_3$ на $6$.

Проще — ориентировать рёбра. Сортируем вершины по степени и ориентируем каждое ребро «вправо» (от меньшей степени к большей, при равных — по номеру). Утверждение: исходящая степень любой вершины $O(\sqrt m)$. Если исходная степень не больше $\sqrt m$, это очевидно. Иначе все вершины правее неё имеют степень не меньше $\sqrt m$, а таких вершин не больше $2m/\sqrt m=O(\sqrt m)$ (–).

Для вершины $u$ отмечаем её исходящих соседей, перебираем исходящее ребро $u\to v$, затем исходящие рёбра $v\to w$ и проверяем отметку $w$. Всего $O(m\sqrt m)$ (–).

// (b) ориентируем каждое ребро от меньшей степени к большей: исходящая степень <= sqrt(2m)
ll triangles_oriented(int n, const vector<pair<int, int>>& edges) {
    vector<int> deg(n, 0); for (auto [u, v] : edges) deg[u]++, deg[v]++;
    auto less_ = [&](int a, int b) { return make_pair(deg[a], a) < make_pair(deg[b], b); };
    vector<vector<int>> out(n);
    for (auto [u, v] : edges) { if (less_(u, v)) out[u].push_back(v); else out[v].push_back(u); }
    ll ans = 0; vector<char> mark(n, 0);
    for (int u = 0; u < n; u++) {
        for (int v : out[u]) mark[v] = 1;
        for (int v : out[u]) for (int w : out[v]) ans += mark[w];
        for (int v : out[u]) mark[v] = 0;
    }
    return ans;
}

«$+x$ всем соседям» и значение в вершине

Граф на $n$ вершинах и $m$ рёбрах; запросы: прибавить $x$ ко всем соседям вершины $v$ и узнать значение в вершине $u$ (–). В зале предлагают корневую по запросам, но лектор хочет другое решение — снова тяжёлые и лёгкие вершины (–).

  • Запрос на лёгкой вершине: честно обходим её соседей (их не больше $B$) и прибавляем $x$ каждому (–).
  • Запрос на тяжёлой вершине: дорого. Заводим массив pend и записываем $x$ в ячейку этой вершины — «отложенное» добавление (–, –).
  • Запрос значения в $u$: к хранимому значению (накопленному от лёгких) добавляем pend[h] по всем тяжёлым $h$, соседним с $u$. Тяжёлых не больше $2m/B$, смежность проверяется по заранее построенной таблице «тяжёлая × вершина» — $\frac{m}{B}\times n$ ячеек (–, –).
struct NeighborAdd {
    int n, B; vector<vector<int>> adj; vector<ll> val, pend; vector<int> hv, hid; vector<vector<char>> adjH;
    NeighborAdd(int n, const vector<pair<int, int>>& edges) : n(n), adj(n), val(n, 0), hid(n, -1) {
        for (auto [u, v] : edges) { adj[u].push_back(v); adj[v].push_back(u); }
        B = isqrt_ceil(2LL * edges.size());
        for (int v = 0; v < n; v++) if ((int)adj[v].size() > B) { hid[v] = hv.size(); hv.push_back(v); }
        pend.assign(hv.size(), 0); adjH.assign(hv.size(), vector<char>(n, 0));
        for (size_t h = 0; h < hv.size(); h++) for (int u : adj[hv[h]]) adjH[h][u] = 1;   // матрица «тяжёлая × все»: корень × n
    }
    void add_neighbors(int v, ll x) {
        if (hid[v] < 0) { for (int u : adj[v]) val[u] += x; }     // лёгкая: пройти по <= B соседям
        else pend[hid[v]] += x;                                   // тяжёлая: только записать отложенную добавку
    }
    ll get(int u) { ll r = val[u]; for (size_t h = 0; h < hv.size(); h++) if (adjH[h][u]) r += pend[h]; return r; }
};

Прыжки по массиву: число путей

Массив $a$; из ячейки $i$ можно прыгнуть в $i+a_i,\ i+2a_i,\ i+3a_i,\dots$ (все прыжки вправо). Нужно число различных множеств посещённых клеток (то есть число путей, начинающихся в клетке $0$; ответ по модулю) (–). Решение за $O(n^2)$ — динамика $dp_i$ («сколько путей приходит в клетку $i$»), в которой из $i$ прыгаем по всем $i+ka_i$; ответ — сумма $dp$ (–). В худшем случае (все $a_i=1$) оно квадратично.

Оценим работу в клетке $i$ как $\frac{n-i}{a_i}\le\frac n{a_i}$. Если $a_i\ge B$, то это $\le n/B$ — такие прыжки делаем честно, всего $O(n\cdot n/B)$ (–). Остаются $a_if[ℓ][r] — сумму $dp_j$ по уже обработанным $j$ с $a_j=\ell$ и $j\bmod\ell=r$ (–). Клетка $i$ получает вклад от всех малых прыжков, суммируя f[ℓ][i mod ℓ] по $\ell–). Время $O(n\sqrt n)$, память $O(n)$ (f[ℓ] имеет длину ℓ) (–).

ll jump_paths(const vector<int>& a, ll mod) {
    int n = a.size(), B = isqrt_ceil(n);
    vector<ll> dp(n, 0); dp[0] = 1;
    vector<vector<ll>> f(B);                                      // f[l][r] — сумма dp[j] по j с a[j] = l, j % l = r
    for (int l = 1; l < B; l++) f[l].assign(l, 0);
    ll ans = 0;
    for (int i = 0; i < n; i++) {
        for (int l = 1; l < B; l++) dp[i] = (dp[i] + f[l][i % l]) % mod;   // малые прыжки «назад»
        ans = (ans + dp[i]) % mod;
        if (a[i] >= B) { for (int j = i + a[i]; j < n; j += a[i]) dp[j] = (dp[j] + dp[i]) % mod; }   // большие — вперёд честно
        else f[a[i]][i % a[i]] = (f[a[i]][i % a[i]] + dp[i]) % mod;
    }
    return ans;
}

Строки и взвешенное расстояние Хэмминга

Строки $s_1,\dots,s_k$ суммарной длины $N$, у каждой вес $w_i$. Запросы: изменить $w_i$ и найти $\sum_{j}w_j\cdot h(s_i,s_j)$ по строкам $j$ той же длины; $h$ — число различающихся позиций (–). Строки разных длин друг на друга не влияют, поэтому задачи по длинам независимы (–). Строк длины больше $\sqrt N$ — не более $\sqrt N$ (–).

Тяжёлые (длинные) строки. Для каждой длины $L>B$ заранее считаем таблицу расстояний между всеми парами строк длины $L$. Каждый символ сравнивается не более чем с $\sqrt N$ строк, поэтому предподсчёт — $O(N\sqrt N)$ (–). Запрос: $\sum_j w_j\,d_{ij}$ по строкам группы, $O(\sqrt N)$; изменение веса — $O(1)$ (–).

Лёгкие (короткие) строки. Для длины $L\le B$ храним $C_{p,c}$ — сумму весов строк с символом $c$ на позиции $p$ — и $D_p=\sum_c C_{p,c}$ (–, –). Расстояние раскладывается по позициям, и для строки $s_i$ нужно сложить по позициям вес всех строк с другим символом: $\sum_p\bigl(D_p-C_{p,s_i[p]}\bigr)$ — $O(L)=O(\sqrt N)$ (–). Изменение веса — пройти по символам своей строки, $O(L)$ (–). Итог — $O\!\left((N+Q)\sqrt N\right)$ (–).

В зале спрашивают про большой алфавит: предподсчёт $C$ требует памяти пропорционально алфавиту, и «сжимать» его нельзя. Лектор решает считать алфавит маленьким (до $26$) (–, –).

struct HammingWeights {
    vector<string> s; vector<ll> w; int B;
    map<int, vector<int>> group;                  // длина -> номера тяжёлых строк
    map<int, vector<vector<int>>> dist;           // длина -> матрица попарных расстояний (только тяжёлые длины)
    vector<int> pos;                              // номер строки внутри своей группы
    map<int, vector<ll>> C, D;                    // лёгкие длины: C[L][p*26+c] и D[L][p]
    HammingWeights(const vector<string>& str, const vector<ll>& weight) : s(str), w(weight), pos(str.size(), -1) {
        ll N = 0; for (auto& x : s) N += x.size();
        B = isqrt_ceil(N);
        for (size_t i = 0; i < s.size(); i++) {
            int L = s[i].size();
            if (L > B) { pos[i] = group[L].size(); group[L].push_back(i); }
            else {
                if (!C.count(L)) { C[L].assign(L * 26, 0); D[L].assign(L, 0); }
                for (int p = 0; p < L; p++) { C[L][p * 26 + s[i][p] - 'a'] += w[i]; D[L][p] += w[i]; }
            }
        }
        for (auto& [L, g] : group) {                   // тяжёлых строк мало, так что матрица небольшая
            dist[L].assign(g.size(), vector<int>(g.size(), 0));
            for (size_t i = 0; i < g.size(); i++) for (size_t j = 0; j < g.size(); j++)
                for (int p = 0; p < L; p++) dist[L][i][j] += s[g[i]][p] != s[g[j]][p];
        }
    }
    void update(int i, ll nw) {
        int L = s[i].size();
        if (L <= B) for (int p = 0; p < L; p++) { C[L][p * 26 + s[i][p] - 'a'] += nw - w[i]; D[L][p] += nw - w[i]; }
        w[i] = nw;
    }
    ll query(int i) {
        int L = s[i].size(); ll r = 0;
        if (L > B) { auto& g = group[L]; for (size_t j = 0; j < g.size(); j++) r += w[g[j]] * dist[L][pos[i]][j]; }
        else for (int p = 0; p < L; p++) r += D[L][p] - C[L][p * 26 + s[i][p] - 'a'];
        return r;
    }
};

Корневая по запросам

Второй ведущий в оставшиеся минуты разгоняется по блоку «корневая по запросам» и предупреждает, что это будет быстро (–). Общий приём для оффлайна: разобьём запросы на блоки по $B$ штук; внутри блока меняется не больше $B$ объектов («плавающие»), остальное — «прибито гвоздями» и обрабатывается один раз на весь блок.

Динамическая связность в оффлайне («ДСУ для бедных»)

Граф с добавлением и удалением рёбер и запросами «лежат ли $u$ и $v$ в одной компоненте» (–). СНМ плохо дружит с произвольными откатами, а «трюк», который сейчас объясняется, превращает любую задачу с удалениями в задачу только с добавлениями (–).

  • Для блока из $B$ запросов выделяем рёбра, которые в нём добавляются или удаляются — не более $B$ штук, плавающие; все остальные живые рёбра присутствуют на всём блоке (–).
  • Остальные рёбра сразу «сжимаем»: запускаем обход (или СНМ) и каждой вершине назначаем номер её компоненты (–). Теперь у нас сжатый граф, в котором только $O(B)$ интересных вершин — концы плавающих рёбер.
  • На запрос обходим в глубину только по плавающим рёбрам, живым в этот момент, начиная с компоненты вершины $u$ ($O(B)$); проверяем, достигли ли компоненты $v$ (–, –).
  • После блока всё забываем и перестраиваем заново: обход по всем рёбрам стоит $O(Q)$ (если граф изначально пуст, то работать нужно только с концами рёбер, а не со всеми вершинами), поэтому при $B=\sqrt Q$ итого $O(Q\sqrt Q)$ (–, –).
vector<char> offline_connectivity(int n, const vector<ConnOp>& ops, int B) {
    set<pair<int, int>> cur; vector<char> ans;
    auto norm = [](int u, int v) { return make_pair(min(u, v), max(u, v)); };
    vector<vector<int>> g(n); vector<int> stamp(n, -1); int cs = 0;
    for (int s = 0; s < (int)ops.size(); s += B) {
        int e = min((int)ops.size(), s + B);
        set<pair<int, int>> touched;
        for (int i = s; i < e; i++) if (ops[i].type != 2) touched.insert(norm(ops[i].u, ops[i].v));
        DSU d(n);
        for (auto& ed : cur) if (!touched.count(ed)) d.unite(ed.first, ed.second);     // «прибитые гвоздями» рёбра
        vector<int> comp(n); for (int v = 0; v < n; v++) comp[v] = d.find(v);
        vector<pair<int, int>> fl;                                                   // плавающие рёбра, живые сейчас
        for (auto& ed : touched) if (cur.count(ed)) fl.push_back(ed);
        for (int i = s; i < e; i++) {
            auto ed = norm(ops[i].u, ops[i].v);
            if (ops[i].type == 0) { cur.insert(ed); fl.push_back(ed); }
            else if (ops[i].type == 1) { cur.erase(ed); fl.erase(find(fl.begin(), fl.end(), ed)); }
            else {
                for (auto& f : fl) { int a = comp[f.first], b = comp[f.second]; g[a].push_back(b); g[b].push_back(a); }
                int src = comp[ops[i].u], dst = comp[ops[i].v]; cs++;
                vector<int> st{src}; stamp[src] = cs;
                while (!st.empty()) { int x = st.back(); st.pop_back(); for (int y : g[x]) if (stamp[y] != cs) { stamp[y] = cs; st.push_back(y); } }
                ans.push_back(stamp[dst] == cs);
                for (auto& f : fl) { g[comp[f.first]].clear(); g[comp[f.second]].clear(); }
            }
        }
    }
    return ans;
}
Проверка. В реализации на каждом блоке строится СНМ по всем вершинам ($O(n+m)$ на блок), без сведения к концам рёбер — так проще и тоже укладывается в $O\!\left(\frac QB(n+m)+QB\right)$. Сверено с перебором (СНМ по рёбрам на каждый запрос) на 200 сериях операций.

Расстояние до ближайшей чёрной клетки

Таблица $n\times m$ ($nm$ порядка $10^6$), все клетки белые; запросы: покрасить клетку в чёрный и найти манхэттенское расстояние от клетки до ближайшей чёрной (–). Условие можно усложнить: чёрные клетки снова становятся белыми — приём тот же (–, –).

Бьём запросы на блоки по $B$. В блоке клетки, меняющие цвет, — «плавающие» (не более $B$ штук), остальные чёрные клетки в этом блоке неизменно чёрные. Запускаем мультиBFS (в очереди сразу все постоянно чёрные клетки, расстояние нуль) за $O(nm)$ (–). Запрос: берём значение BFS и просматриваем в лоб плавающие клетки, живые (чёрные) в данный момент, считая $|x_1-x_2|+|y_1-y_2|$ (–). Время $O\!\left(\frac QB\,nm+QB\right)$; значение $B$ подбирается из равенства (–).

// тип 0 — покрасить в чёрный, 1 — вернуть белый, 2 — спросить; -1, если чёрных клеток нет
vector<int> grid_nearest(int n, int m, const vector<array<int, 3>>& ops, int B) {
    vector<char> black(n * m, 0); vector<int> ans; const int INF = 1e9;
    for (int s = 0; s < (int)ops.size(); s += B) {
        int e = min((int)ops.size(), s + B);
        vector<char> touched(n * m, 0); vector<int> cells;
        for (int i = s; i < e; i++) if (ops[i][0] != 2) { int c = ops[i][1] * m + ops[i][2]; if (!touched[c]) { touched[c] = 1; cells.push_back(c); } }
        vector<int> dist(n * m, INF); queue<int> q;
        for (int c = 0; c < n * m; c++) if (black[c] && !touched[c]) { dist[c] = 0; q.push(c); }      // мультиBFS из «прибитых» чёрных
        while (!q.empty()) {
            int c = q.front(); q.pop(); int x = c / m, y = c % m;
            const int dx[4] = {1, -1, 0, 0}, dy[4] = {0, 0, 1, -1};
            for (int k = 0; k < 4; k++) {
                int nx = x + dx[k], ny = y + dy[k];
                if (nx < 0 || ny < 0 || nx >= n || ny >= m || dist[nx * m + ny] != INF) continue;
                dist[nx * m + ny] = dist[c] + 1; q.push(nx * m + ny);
            }
        }
        for (int i = s; i < e; i++) {
            int c = ops[i][1] * m + ops[i][2];
            if (ops[i][0] == 0) black[c] = 1;
            else if (ops[i][0] == 1) black[c] = 0;
            else {
                int best = dist[c];
                for (int f : cells) if (black[f]) best = min(best, abs(f / m - ops[i][1]) + abs(f % m - ops[i][2]));   // плавающие — в лоб
                ans.push_back(best >= INF ? -1 : best);
            }
        }
    }
    return ans;
}
Проверка. Покраска и обратная перекраска сверены с прямым перебором всех чёрных клеток на 200 сериях (сетки до $7\times7$, блоки от $1$ до $8$). Если чёрных клеток нет, возвращается $-1$.

Вес минимального остова при добавлении и удалении рёбер

Рёбра добавляются и удаляются, нужен вес остова после запроса (–). Лектор предупреждает: «задача скорее идейная — про устройство остовов, а не про корневую» (–). Блок запросов длины $B$; рёбра, меняющиеся в блоке, — плавающие (множество $D$, $|D|\le B$); остальные живые рёбра — множество $E$, они живы на всём блоке (–).

  1. Строим остов (Краскал) на всех рёбрах $E\cup D$, причём в порядке возрастания веса (–). В нём не больше $B$ плавающих рёбер, поэтому хотя бы $n-1-B$ рёбер — из $E$ (–). Если остов не связен, то на всём блоке ответ «$-1$».
  2. Утверждение. Неплавающие рёбра этого остова входят в минимальный остов в любой момент блока (–). Лектор оставляет доказательство на потом и просит поверить; в записи оно не рассказывается (–, –).
  3. Сжимаем граф по этим «вечным» рёбрам: остаётся не больше $B+1$ компонент (–).
  4. Краскал продолжается на сжатом графе только по неплавающим рёбрам: выбирается не более $B$ рёбер (–). Остальные неплавающие рёбра вообще не понадобятся.
  5. На запрос остаётся не более $B$ выбранных неплавающих рёбер плюс не более $B$ живых плавающих. Сортируем их, запускаем Краскала по сжатому графу и добавляем вес «вечных» рёбер: $O(B\log B)$ (–, –).

Почему этого достаточно: минимальный остов при добавлении ребра меняется лишь заменой наибольшего ребра на цикле (свойство «минимакс на пути»); мысленно добавляя плавающие рёбра по одному, мы получаем настоящий остов, уже лежащий в выбранных рёбрах (–).

Предподсчёт блока — Краскал по $O(Q)$ рёбрам, $O(Q\log Q)$; блоков $Q/B$. Итого $O\!\left(\frac QB\,Q\log Q+QB\log B\right)$, при $B=\sqrt Q$ — $O(Q\sqrt Q\log Q)$ (–). Замена СНМ перебором «за $B$» дала бы $B^3$ на запрос и была бы хуже (–).

vector<ll> offline_mst(int n, const vector<MstOp>& ops, int B) {
    map<int, MEdge> cur; vector<ll> ans;
    auto cmp = [](const MEdge& a, const MEdge& b) { return make_pair(a.w, a.id) < make_pair(b.w, b.id); };
    for (int s = 0; s < (int)ops.size(); s += B) {
        int e = min((int)ops.size(), s + B);
        set<int> touched; vector<MEdge> all;
        for (auto& kv : cur) all.push_back(kv.second);
        for (int i = s; i < e; i++) if (ops[i].type != 2) {
            touched.insert(ops[i].id);
            if (ops[i].type == 0) all.push_back({ops[i].id, ops[i].u, ops[i].v, ops[i].w});
        }
        sort(all.begin(), all.end(), cmp);
        // шаг А: остов на всех рёбрах, живых хоть когда-то в блоке; неплавающие рёбра остова лежат в остове всегда
        DSU d(n); vector<MEdge> forced; ll wForced = 0; int comps = n;
        for (auto& ed : all) if (d.unite(ed.u, ed.v)) { comps--; if (!touched.count(ed.id)) { forced.push_back(ed); wForced += ed.w; } }
        bool spanning = comps == 1;
        vector<MEdge> cand; vector<int> cid(n); int C = 0;
        if (spanning) {
            DSU d2(n); for (auto& ed : forced) d2.unite(ed.u, ed.v);        // сжимаем граф по ним
            vector<int> label(n, -1);
            for (int v = 0; v < n; v++) { int r = d2.find(v); if (label[r] < 0) label[r] = C++; cid[v] = label[r]; }
            // шаг Б: из остальных неплавающих рёбер оставляем только те, что нужны остову сжатого графа
            DSU d3(C);
            for (auto& ed : all) if (!touched.count(ed.id) && d3.unite(cid[ed.u], cid[ed.v])) cand.push_back(ed);
        }
        set<int> alive;                                                      // плавающие рёбра, живые прямо сейчас
        for (int id : touched) if (cur.count(id)) alive.insert(id);
        map<int, MEdge> fl; for (int i = s; i < e; i++) if (ops[i].type == 0) fl[ops[i].id] = {ops[i].id, ops[i].u, ops[i].v, ops[i].w};
        for (int id : alive) fl[id] = cur[id];
        for (int i = s; i < e; i++) {
            if (ops[i].type == 0) { cur[ops[i].id] = fl[ops[i].id]; alive.insert(ops[i].id); }
            else if (ops[i].type == 1) { cur.erase(ops[i].id); alive.erase(ops[i].id); }
            else {
                if (!spanning) { ans.push_back(-1); continue; }
                vector<MEdge> es = cand; for (int id : alive) es.push_back(fl[id]);
                sort(es.begin(), es.end(), cmp);                              // не больше 2B рёбер на запрос
                DSU dq(C); ll tot = wForced; int cc = C;
                for (auto& ed : es) if (dq.unite(cid[ed.u], cid[ed.v])) { tot += ed.w; cc--; }
                ans.push_back(cc == 1 ? tot : -1);
            }
        }
    }
    return ans;
}
Проверка. Ребро имеет идентификатор и вес; порядок рёбер при равных весах задан парой (вес, номер) — это важно для утверждения о «вечных» рёбрах. Сверено с пересчётом остова с нуля на каждый запрос на 300 сериях операций (в том числе, когда граф несвязен). Перебор значений $\lfloor x/i\rfloor$

Последний приём, «с которым на олимпиадах встречались три раза»: перебрать все различные значения вида $\lfloor x/i\rfloor$, $i=1,2,\dots$ (–). Различных значений не больше $2\sqrt x$: при $i\le\sqrt x$ — не более $\sqrt x$ значений, при $i>\sqrt x$ значение меньше $\sqrt x$ (–).

Идём по возрастанию $i$ и ищем минимальное $j>i$, у которого значение меньше: если $k=\lfloor x/i\rfloor$, то условие $\lfloor x/j\rfloorx/k$. Наименьшее такое целое $j$ — это $\lfloor x/k\rfloor+1$ (–). Цикл: for (i = 1; i <= x; i = x / (x / i) + 1). Если округление вверх — упражнение (–).

vector<array<ll, 3>> floor_blocks(ll x) {                  // {значение k, i от, i до}
    vector<array<ll, 3>> r;
    for (ll i = 1; i <= x;) { ll k = x / i, j = x / k; r.push_back({k, i, j}); i = j + 1; }   // следующий i — это x/k + 1
    return r;
}
